    // Stats Counter Animation
    function animateCounters() {
      document.querySelectorAll('.stat-item h4').forEach(el => {
        const target = parseInt(el.textContent.replace(/[^0-9]/g, ''));
        let count = 0;
        const duration = 2000;
        const increment = target / (duration / 16);
        
        const update = () => {
          count += increment;
          if (count < target) {
            el.innerHTML = Math.floor(count) + (el.innerHTML.includes('s') ? '<span>s</span>' : el.innerHTML.includes('+') ? '<span>+</span>' : '');
            requestAnimationFrame(update);
          } else {
            el.innerHTML = target + (el.innerHTML.includes('s') ? '<span>s</span>' : el.innerHTML.includes('+') ? '<span>+</span>' : '');
          }
        };
        update();
      });
    }

      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
          if (entry.target.classList.contains('stats-strip')) animateCounters();
        }
      });
    }, { threshold: 0.1 });

    document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
