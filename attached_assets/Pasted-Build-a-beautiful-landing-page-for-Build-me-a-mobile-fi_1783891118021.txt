        appState.isMenuOpen = !appState.isMenuOpen;
        menu.classList.toggle('open', appState.isMenuOpen);
        document.body.style.overflow = appState.isMenuOpen ? 'hidden' : '';
      };

      window.addToCart = function(id, name, price) {
        const existing = appState.cart.find(item => item.id === id);
        if (existing) {
          existing.qty++;
        } else {
          appState.cart.push({ id, name, price, qty: 1 });
        }
        renderCart();
        if(!appState.isCartOpen) window.toggleCart();
        showToast(`Added ${name} to bag`);
      };

      window.updateQty = function(id, delta) {
        const item = appState.cart.find(item => item.id === id);
        if (item) {
          item.qty += delta;
          if (item.qty <= 0) {
            appState.cart = appState.cart.filter(i => i.id !== id);
          }
        }
        renderCart();
      };

      window.checkout = function() {
        if (appState.cart.length === 0) return showToast("Your bag is empty");
        const total = appState.cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
        const paypalUrl = `https://paypal.me/bloomandco/${total.toFixed(2)}`;
        window.open(paypalUrl, '_blank');
      };

      window.openLegal = function(title, content) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = `<p>${content}</p>`;
        document.getElementById('legalModal').style.display = 'flex';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
      };

      function renderCart() {
        const container = document.getElementById('cart-items');
        const countBadge = document.getElementById('cart-count');
        const totalEl = document.getElementById('cart-total');
        
        const totalItems = appState.cart.reduce((sum, item) => sum + item.qty, 0);
        const totalPrice = appState.cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
        
        countBadge.textContent = totalItems;
        totalEl.textContent = `$${totalPrice.toFixed(2)}`;
        
        if (appState.cart.length === 0) {
          container.innerHTML = `
            <div class="flex flex-col items-center justify-center h-full text-center py-20 opacity-40">
              <i data-lucide="shopping-bag" class="w-12 h-12 mb-4"></i>
              <p class="text-sm uppercase tracking-widest font-bold">Your bag is empty</p>
            </div>
          `;
          lucide.createIcons();
          return;
        }
        
        container.innerHTML = appState.cart.map(item => `
          <div class="flex gap-4 items-center group">
            <div class="w-20 h-24 bg-brand-text/5 rounded-sm overflow-hidden flex-shrink-0">
              <img src="https://images.pexels.com/photos/${getImageId(item.id)}/pexels-photo-${getImageId(item.id)}.jpeg?auto=compress&cs=tinysrgb&w=200" 
                   class="w-full h-full object-cover" onerror="this.style.display='none'">
            </div>
            <div class="flex-1">
              <h4 class="font-heading text-lg mb-1">${item.name}</h4>
              <p class="text-brand-text/40 text-xs uppercase tracking-widest mb-3">$${item.price.toFixed(2)}</p>
              <div class="flex items-center gap-4">
                <div class="flex items-center border border-brand-text/10 rounded-full px-3 py-1">
                  <button onclick="window.updateQty(${item.id}, -1)" class="hover:text-brand-accent transition-colors"><i data-lucide="minus" class="w-3 h-3"></i></button>
                  <span class="mx-4 text-xs font-bold">${item.qty}</span>
                  <button onclick="window.updateQty(${item.id}, 1)" class="hover:text-brand-accent transition-colors"><i data-lucide="plus" class="w-3 h-3"></i></button>
                </div>
              </div>
            </div>
            <div class="text-right">
              <p class="font-bold text-sm">$${(item.price * item.qty).toFixed(2)}</p>
            </div>
          </div>
        `).join('');
        lucide.createIcons();
      }

      function getImageId(id) {
        const ids = { 1: 7195133, 2: 6707628, 3: 6634653, 4: 3951746 };
        return ids[id] || 6707631;
      }

      function showToast(msg) {
        const toast = document.createElement('div');
        toast.className = 'fixed bottom-8 left-1/2 -translate-x-1/2 bg-brand-dark text-white px-6 py-3 rounded-sm text-xs font-bold uppercase tracking-widest shadow-2xl z-[200] animate-bounce';
        toast.textContent = msg;
        document.body.appendChild(toast);
        setTimeout(() => toast.remove(), 2500);
      }

      // Initial Setup
      document.getElementById('yr').textContent = new Date().getFullYear();
      lucide.createIcons();
      
      // Reveal Animation
      const observer = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });
      
      document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
      
      // Safety net for reveal
      setTimeout(() => {
        document.querySelectorAll('.reveal:not(.visible)').forEach(el => {
          el.classList.add('visible');
        });
      }, 1500);

    })();
  </script>

  <script>
    (function() {
      const appState = {
        cart: [],
        isCartOpen: false,
        isMenuOpen: false,
        products: [
          { id: 1, name: 'Lavender Dream', price: 29 }, // Updated price
          { id: 2, name: 'Sandalwood Noir', price: 32 },
          { id: 3, name: 'Rose Petal', price: 28 },
          { id: 4, name: 'Vanilla Sky', price: 26 },
          { id: 5, name: 'Midnight Oud', price: 38 } // Added product
        ]
      };

      window.toggleCart = function() {
        const drawer = document.getElementById('cart-drawer');
        const overlay = document.getElementById('cart-overlay');
        appState.isCartOpen = !appState.isCartOpen;
        
        if (appState.isCartOpen) {
          drawer.classList.add('open');
          overlay.classList.remove('hidden');
          document.body.style.overflow = 'hidden';
        } else {
          drawer.classList.remove('open');
          overlay.classList.add('hidden');
          document.body.style.overflow = '';
        }
      };

      window.toggleMenu = function() {
        const menu = document.getElementById('mobile-nav');
        appState.isMenuOpen = !appState.isMenuOpen;
        menu.classList.toggle('open', appState.isMenuOpen);
        document.body.style.overflow = appState.isMenuOpen ? 'hidden' : '';
      };

      window.addToCart = function(id, name, price) {
        const existing = appState.cart.find(item => item.id === id);
        if (existing) {
          existing.qty++;
        } else {
          appState.cart.push({ id, name, price, qty: 1 });
        }
        renderCart();
        if(!appState.isCartOpen) window.toggleCart();
        showToast(`Added ${name} to bag`);
      };

      window.updateQty = function(id, delta) {
        const item = appState.cart.find(item => item.id === id);
        if (item) {
          item.qty += delta;
          if (item.qty <= 0) {
            appState.cart = appState.cart.filter(i => i.id !== id);
          }
        }
        renderCart();
      };

      window.checkout = function() {
        if (appState.cart.length === 0) return showToast("Your bag is empty");
        const total = appState.cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
        const paypalUrl = `https://paypal.me/bloomco/${total.toFixed(2)}`; // Updated PayPal link
        window.open(paypalUrl, '_blank');
      };

      window.openLegal = function(title, content) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = `<p>${content}</p>`;
        document.getElementById('legalModal').style.display = 'flex';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
      };

      function renderCart() {
        const container = document.getElementById('cart-items');
        const countBadge = document.getElementById('cart-count');
        const totalEl = document.getElementById('cart-total');
        
        const totalItems = appState.cart.reduce((sum, item) => sum + item.qty, 0);
        const totalPrice = appState.cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
        
        countBadge.textContent = totalItems;
        totalEl.textContent = `$${totalPrice.toFixed(2)}`;
        
        if (appState.cart.length === 0) {
          container.innerHTML = `
            <div class="flex flex-col items-center justify-center h-full text-center py-20 opacity-40">
              <i data-lucide="shopping-bag" class="w-12 h-12 mb-4"></i>
              <p class="text-sm uppercase tracking-widest font-bold">Your bag is empty</p>
            </div>
          `;
          lucide.createIcons();
          return;
        }
        
        container.innerHTML = appState.cart.map(item => `
          <div class="flex gap-4 items-center group">
            <div class="w-20 h-24 bg-brand-text/5 rounded-sm overflow-hidden flex-shrink-0">
              <img src="https://images.pexels.com/photos/${getImageId(item.id)}/pexels-photo-${getImageId(item.id)}.jpeg?auto=compress&cs=tinysrgb&w=200" 
                   class="w-full h-full object-cover" onerror="this.style.display='none'">
            </div>
            <div class="flex-1">
              <h4 class="font-heading text-xl mb-1">${item.name}</h4>
              <p class="text-brand-text/40 text-xs uppercase tracking-widest mb-3">$${item.price.toFixed(2)}</p>
              <div class="flex items-center gap-4">
                <div class="flex items-center border border-brand-text/10 rounded-full px-3 py-1">
                  <button onclick="window.updateQty(${item.id}, -1)" class="hover:text-brand-accent transition-colors"><i data-lucide="minus" class="w-3 h-3"></i></button>
                  <span class="mx-4 text-xs font-bold">${item.qty}</span>
                  <button onclick="window.updateQty(${item.id}, 1)" class="hover:text-brand-accent transition-colors"><i data-lucide="plus" class="w-3 h-3"></i></button>
                </div>
              </div>
            </div>
            <div class="text-right">
              <p class="font-bold text-sm">$${(item.price * item.qty).toFixed(2)}</p>
            </div>
          </div>
        `).join('');
        lucide.createIcons();
      }

      function getImageId(id) {
        const ids = { 1: 7195133, 2: 6707628, 3: 6634653, 4: 3951746, 5: 4464482 };
        return ids[id] || 6707631;
      }

      function showToast(msg) {
        const toast = document.createElement('div');
        toast.className = 'fixed bottom-8 left-1/2 -translate-x-1/2 bg-brand-dark text-white px-6 py-3 rounded-sm text-xs font-bold uppercase tracking-widest shadow-2xl z-[200] animate-bounce';
        toast.textContent = msg;
        document.body.appendChild(toast);
        setTimeout(() => toast.remove(), 2500);
      }

      document.getElementById('yr').textContent = new Date().getFullYear();
      lucide.createIcons();
      
      const observer = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });
      
      document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
      
      setTimeout(() => {
        document.querySelectorAll('.reveal:not(.visible)').forEach(el => {
          el.classList.add('visible');
        });
      }, 1500);

    })();
  </script>
