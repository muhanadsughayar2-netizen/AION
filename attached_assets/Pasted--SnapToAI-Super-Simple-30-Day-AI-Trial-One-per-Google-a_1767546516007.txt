// SnapToAI - Super Simple 30-Day AI Trial (One per Google account / sync storage)
// Made fun & easy! 🚀✨

const TRIAL_DAYS = 30;
const TRIAL_KEY = 'snapMagicTrial'; // Cute name for the storage key 😏

function checkTrial() {
  chrome.storage.sync.get(TRIAL_KEY, (result) => {
    const trial = result[TRIAL_KEY];

    // No trial yet? Party time!
    if (!trial || !trial.startDate) {
      console.log("🎉 First time! You can start your 30-day AI magic!");
      return { canUseAI: true, isTrial: true, message: "Your 30-day AI adventure starts now!" };
    }

    const daysUsed = Math.floor((Date.now() - trial.startDate) / (1000 * 60 * 60 * 24));

    if (daysUsed < TRIAL_DAYS) {
      const daysLeft = TRIAL_DAYS - daysUsed;
      console.log(`🔮 AI Magic still active! ${daysLeft} days of fun left!`);
      return { canUseAI: true, isTrial: true, daysLeft };
    } else {
      console.log("😢 Trial time over... time to unlock the full magic!");
      return { canUseAI: false, isTrial: false, message: "Your 30-day trial has ended. Subscribe for unlimited snaps! 💜" };
    }
  });
}

function startTheMagicTrial() {
  chrome.storage.sync.get(TRIAL_KEY, (result) => {
    if (result[TRIAL_KEY]) {
      // Already had a trial — no double dipping!
      alert("You've already enjoyed your 30-day AI trial! ✨\nSubscribe to keep the magic going!");
      return;
    }

    // Let's start the party!
    const trialData = {
      startDate: Date.now(),
      started: true
    };

    chrome.storage.sync.set({ [TRIAL_KEY]: trialData }, () => {
      alert("🎊 Boom! Your 30-day AI trial is LIVE!\nGo snap some charts and have fun!");
      // Here you can enable AI features, show a cool welcome animation, etc.
    });
  });
}

// When extension is installed or updated — check the vibe
chrome.runtime.onInstalled.addListener(() => {
  checkTrial();
});

// Optional: Check every time the browser starts (nice touch)
chrome.runtime.onStartup.addListener(() => {
  checkTrial();
});

// How to use it in your popup or content script:
// Just call startTheMagicTrial() when user first tries an AI feature
// And use checkTrial() to decide if AI should work
