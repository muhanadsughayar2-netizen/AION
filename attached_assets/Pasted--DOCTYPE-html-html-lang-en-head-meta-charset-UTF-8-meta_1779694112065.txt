  <script>
    // 1. Reveal Animations
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
        }
      });
    }, { threshold: 0.1 });

    document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

    // 2. Interactive Bio-Audit Score Logic
    let currentScore = 0;
    const scoreVal = document.getElementById('score-val');
    const gaugeFill = document.getElementById('gauge-fill');
    const circumference = 2 * Math.PI * 100;

    document.querySelectorAll('.food-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        const val = parseInt(btn.getAttribute('data-value'));
        
        if (btn.classList.contains('active')) {
          btn.classList.remove('active');
          currentScore -= val;
        } else {
          btn.classList.add('active');
          currentScore += val;
        }

        updateScore(currentScore);
      });
    });

    function updateScore(score) {
      const visualScore = Math.min(Math.max(score, 0), 100);
      
      const start = parseInt(scoreVal.textContent);
      const duration = 600;
      const startTime = performance.now();

      function animate(now) {
        const elapsed = now - startTime;
        const progress = Math.min(elapsed / duration, 1);
        const currentNum = Math.floor(start + (score - start) * progress);
        scoreVal.textContent = currentNum;

        if (progress < 1) requestAnimationFrame(animate);
      }
      requestAnimationFrame(animate);

      const offset = circumference - (visualScore / 100) * circumference;
      gaugeFill.style.strokeDashoffset = offset;
    }

    // 3. FIX: Smooth Scrolling with Header Offset
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
      anchor.addEventListener('click', function (e) {
        e.preventDefault();
        const targetId = this.getAttribute('href');
        const targetElement = document.querySelector(targetId);
        
        if (targetElement) {
          const headerOffset = 90; // Height of your fixed nav
          const elementPosition = targetElement.getBoundingClientRect().top;
          const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
  
          window.scrollTo({
            top: offsetPosition,
            behavior: "smooth"
          });
        }
      });
    });

    // 4. FIX: Safe Parallax
    window.addEventListener('scroll', () => {
      const scroll = window.pageYOffset;
      const img = document.querySelector('.hero-image');
      if (img) {
        img.style.transform = `translateY(${scroll * 0.15}px)`;
      }
    });
    
    // 5. FIX: Make Pricing Buttons Work
    document.querySelectorAll('.price-card .btn').forEach(btn => {
       btn.addEventListener('click', (e) => {
           e.preventDefault();
           // Find the audit section and scroll to it
           const targetElement = document.querySelector('#audit');
           const headerOffset = 90;
           const elementPosition = targetElement.getBoundingClientRect().top;
           const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
           
           window.scrollTo({
             top: offsetPosition,
             behavior: "smooth"
           });
           
           // Highlight the audit box briefly to show them where they are

           }, 2000);
       });
    });
