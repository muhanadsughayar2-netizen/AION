// HIGH-RISK ACTION DETECTION
const HIGH_RISK_PATTERNS = /pay|purchase|buy|delete|remove|checkout|confirm\s*order|submit\s*payment|cancel\s*account/i;

function isHighRiskElement(el) {
  if (!el) return false;
  const text = (el.innerText || el.textContent || el.getAttribute('aria-label') || el.title || el.value || '').toLowerCase();
  return HIGH_RISK_PATTERNS.test(text);
}

// AGENT AUTOMATION ACTION HANDLER
async function handleAgentAction(action, params) {
  createGhostCursor();
  if (action === 'click' || action === 'locateForClick') {
    let element = document.querySelector(params.selector);
    if (!element && params.text) {
      const all = querySelectorAllShadow('button, a, span, div[role="button"], input[type="button"]');
      for (const el of all) {
        if ((el.textContent || '').toLowerCase().includes(params.text.toLowerCase())) {
          element = el; break;
        }
      }
    }
    if (!element) throw new Error(`Element not found for "${params.text || params.selector}"`);

    const highRisk = isHighRiskElement(element);
    element.scrollIntoView({ behavior: 'smooth', block: 'center' });
    const center = getElementCenter(element);
    moveGhostCursor(center.x, center.y);
    highlightElement(element);

    if (action === 'locateForClick') {
      return { success: true, x: center.x, y: center.y, isHighRisk: highRisk };
    }
    element.click();
    return { success: true, isHighRisk: highRisk };
  }
}

class CdpSessionPool {
  constructor() {
    this.sessions = new Map();
    this.locks    = new Map();
  }

  async acquire(tabId) {
    let releaseLock;
    const prev = this.locks.get(tabId) || Promise.resolve();
    const next  = new Promise(r => { releaseLock = r; });
    this.locks.set(tabId, prev.then(() => next));
    await prev;

    try {
      let s = this.sessions.get(tabId);
      if (!s) { s = { count: 0, timeout: null, attached: false }; this.sessions.set(tabId, s); }
      if (s.timeout) { clearTimeout(s.timeout); s.timeout = null; }
      if (!s.attached) {
        await chrome.debugger.attach({ tabId }, '1.3');
        s.attached = true;
      }
      s.count++;

      const send = (method, params) => new Promise((resolve, reject) => {
        chrome.debugger.sendCommand({ tabId }, method, params || {}, result => {
          if (chrome.runtime.lastError) reject(new Error(chrome.runtime.lastError.message));
          else resolve(result);
        });
      });

      const release = () => {
        s.count = Math.max(0, s.count - 1);
        if (s.count === 0) {
          s.timeout = setTimeout(async () => {
            if (s.count === 0 && s.attached) {
              try { await chrome.debugger.detach({ tabId }); } catch (_) {}
              s.attached = false;
              this.sessions.delete(tabId);
            }
          }, 45000);
        }
        releaseLock();
      };

      return { send, release };
    } catch (err) {
      releaseLock();
      throw err;
    }
  }
}

async function getCouncilConsensus({ chatgpt = '', claude = '', grok = '' } = {}) {
  const keyResult = await chrome.storage.sync.get(['geminiApiKey']);
  const apiKey = keyResult.geminiApiKey;
  if (!apiKey) return null;

  const judgePrompt =
    `Act as a Supreme Court Justice synthesizing AI opinions. You have received three responses:\n\n` +
    `**ChatGPT:** ${chatgpt}\n` +
    `**Claude:** ${claude}\n` +
    `**Grok:** ${grok}\n\n` +
    `Produce a Consensus Report with three sections:\n` +
    `## ✅ Consensus — The Truth\n` +
    `## ⚠️ Contradictions — The Risks\n` +
    `## 🏆 Final Recommendation`;

  const resp = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`,
    {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ role: 'user', parts: [{ text: judgePrompt }] }] })
    }
  );
  const data = await resp.json();
  return data?.candidates?.[0]?.content?.parts?.[0]?.text || null;
}
