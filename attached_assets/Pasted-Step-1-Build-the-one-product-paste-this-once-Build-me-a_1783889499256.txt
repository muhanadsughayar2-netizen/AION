  <script>
    (function() {
      // 1. Initial Data
      const PRODUCTS = [
        { id: 1, name: "Silver Moon Ring", price: 45, img: "https://images.pexels.com/photos/2657594/pexels-photo-2657594.jpeg?auto=compress&cs=tinysrgb&w=800", desc: "Forged in the glow of the lunar cycle." },
        { id: 2, name: "Crystal Pendant", price: 55, img: "https://images.pexels.com/photos/6415712/pexels-photo-6415712.jpeg?auto=compress&cs=tinysrgb&w=800", desc: "A fragment of starlight for your journey." },
        { id: 3, name: "Gold Ear Cuff", price: 28, img: "https://images.pexels.com/photos/1460838/pexels-photo-1460838.jpeg?auto=compress&cs=tinysrgb&w=800", desc: "Minimalist elegance for the modern mystic." },
        { id: 4, name: "Rose Quartz Bracelet", price: 42, img: "https://images.pexels.com/photos/2853909/pexels-photo-2853909.jpeg?auto=compress&cs=tinysrgb&w=800", desc: "Woven with intentions of love and light." }
      ];

      const appState = {
        cart: JSON.parse(localStorage.getItem('luna_cart') || '[]'),
        
        updateCart(product) {
          const existing = this.cart.find(item => item.id === product.id);
          if (existing) {
            existing.qty += 1;
          } else {
            this.cart.push({ ...product, qty: 1 });
          }
          this.saveAndRender();
          window.showToast(`Added ${product.name} to bag`);
        },

        removeFromCart(id) {
          this.cart = this.cart.filter(item => item.id !== id);
          this.saveAndRender();
        },

        saveAndRender() {
          localStorage.setItem('luna_cart', JSON.stringify(this.cart));
          this.render();
        },

        render() {
          const grid = document.getElementById('product-grid');
          const cartList = document.getElementById('cart-items');
          const cartCount = document.getElementById('cart-count');
          const cartTotal = document.getElementById('cart-total');
          const emptyMsg = document.getElementById('empty-cart');

          // Render Products
          if (grid && grid.children.length === 0) {
            PRODUCTS.forEach(p => {
              const el = document.createElement('div');
              el.className = 'product-card reveal group';
              el.innerHTML = `
                <div class="relative aspect-[4/5] overflow-hidden bg-luna-cream mb-6">
                  <img src="${p.img}" alt="${p.name}" class="w-full h-full object-cover" onerror="this.parentElement.style.background='#1A1A2E'">
                  <button onclick="window.addToCart(${p.id})" class="absolute bottom-4 left-4 right-4 bg-white py-4 text-xs font-bold uppercase tracking-widest translate-y-12 opacity-0 group-hover:translate-y-0 group-hover:opacity-100 transition-all duration-300 btn-active-scale shadow-xl">
                    Add to Bag — $${p.price}
                  </button>
                </div>
                <h3 class="font-heading text-xl mb-1 font-bold">${p.name}</h3>
                <p class="text-xs text-luna-muted uppercase tracking-widest mb-2">$${p.price}.00</p>
                <p class="text-sm text-luna-muted font-light">${p.desc}</p>
              `;
              grid.appendChild(el);
            });
          }

          // Render Cart
          const totalItems = this.cart.reduce((acc, item) => acc + item.qty, 0);
          if (cartCount) {
            cartCount.textContent = totalItems;
            cartCount.style.opacity = totalItems > 0 ? '1' : '0';
          }

          if (cartList) {
            if (this.cart.length === 0) {
              cartList.innerHTML = '';
              if (emptyMsg) cartList.appendChild(emptyMsg);
              if (cartTotal) cartTotal.textContent = '$0.00';
            } else {
              if (emptyMsg) emptyMsg.remove();
              cartList.innerHTML = this.cart.map(item => `
                <div class="flex gap-4 animate-in fade-in slide-in-from-right-4 duration-300">
                  <div class="w-20 h-24 bg-luna-cream flex-shrink-0">
                    <img src="${item.img}" class="w-full h-full object-cover" onerror="this.parentElement.style.background='#1A1A2E'">
                  </div>
                  <div class="flex-1">
                    <div class="flex justify-between mb-1">
                      <h4 class="font-heading text-lg font-bold">${item.name}</h4>
                      <button onclick="window.removeFromCart(${item.id})" class="text-luna-muted hover:text-red-500 transition-colors">
                        <i data-lucide="trash-2" class="w-4 h-4"></i>
                      </button>
                    </div>
                    <p class="text-xs text-luna-muted mb-4 uppercase tracking-widest">$${item.price}.00 &times; ${item.qty}</p>
                  </div>
                </div>
              `).join('');
              
              const total = this.cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
              if (cartTotal) cartTotal.textContent = `$${total}.00`;
              lucide.createIcons();
            }
          }
        }
      };

      // 2. Window Functions
      window.addToCart = (id) => {
        const product = PRODUCTS.find(p => p.id === id);
        appState.updateCart(product);
      };

      window.removeFromCart = (id) => {
        appState.removeFromCart(id);
      };

      window.toggleCart = () => {
        const drawer = document.getElementById('cart-drawer');
        const overlay = document.getElementById('cart-overlay');
        const isOpen = drawer.classList.toggle('open');
        overlay.style.opacity = isOpen ? '1' : '0';
        overlay.style.pointerEvents = isOpen ? 'auto' : 'none';
        document.body.style.overflow = isOpen ? 'hidden' : '';
      };

      window.openLegal = (title, body) => {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = body;
        document.getElementById('legalModal').style.display = 'flex';
      };

      window.closeLegal = () => {
        document.getElementById('legalModal').style.display = 'none';
      };

      window.handleSubscribe = (e) => {
        e.preventDefault();
        const success = document.getElementById('subscribe-success');
        success.classList.remove('hidden');
        e.target.reset();
        setTimeout(() => success.classList.add('hidden'), 5000);
      };

      window.showToast = function(msg) {
        const toast = document.getElementById('toast');
        toast.textContent = msg;
        toast.style.opacity = '1';
        toast.style.transform = 'translate(-50%, -24px)';
        setTimeout(() => {
          toast.style.opacity = '0';
          toast.style.transform = 'translate(-50%, 16px)';
        }, 2500);
      }

      // 3. Scroll Effects
      window.addEventListener('scroll', () => {
        const img = document.getElementById('hero-img');
        if (img) {
          const scroll = window.pageYOffset;
          img.style.transform = `translateY(${scroll * 0.4}px)`;
        }
      });

      // 4. Initialization
      lucide.createIcons();
      appState.render();

      const _io = new IntersectionObserver(entries => {
        entries.forEach(e => {
          if (e.isIntersecting) {
            e.target.classList.add('visible');
            _io.unobserve(e.target);
          }
        });
      }, { threshold: 0.1 });

      document.querySelectorAll('.reveal').forEach(el => _io.observe(el));
      
      // Safety net
      setTimeout(() => {
        document.querySelectorAll('.reveal').forEach(el => {
          el.style.opacity = '1';
          el.style.transform = 'none';
        });
      }, 1500);

    })();
  </script>
