// HARDENED PERSISTENT CDP SESSION POOL (Self-Healing & MV3 Compliant)
// ============================================================================
class CdpSessionPool {
  constructor() {
    this.sessions = new Map(); // tabId -> { attached, refCount }
    this.mutexes = new Map();  // tabId -> Promise (for serializing commands)
    
    // Listen for manual debugger detaches (e.g. user clicks "Cancel" on the Chrome infobar)
    chrome.debugger.onEvent.addListener(this.handleDebuggerEvent.bind(this));
    chrome.debugger.onDetach.addListener(this.handleDebuggerDetach.bind(this));
  }

  handleDebuggerEvent(source, method, params) {
    // Process sub-session attachments cleanly
    if (method === 'Target.attachedToTarget') {
      const { sessionId, targetInfo } = params || {};
      if (sessionId && targetInfo) {
        chrome.debugger.getTargets(targets => {
          const match = (targets || []).find(t => t.id === targetInfo.targetId);
          if (match?.tabId) {
            console.log(`[Aion] Sub-session resolved: tabId=${match.tabId} url=${targetInfo.url}`);
          }
        });
      }
    }
  }

  handleDebuggerDetach(source) {
    const { tabId } = source;
    if (tabId && this.sessions.has(tabId)) {
      console.warn(`[Aion] Debugger manually detached from tab ${tabId}`);
      this.sessions.delete(tabId);
    }
  }

  async acquire(tabId) {
    // 1. Serialize access to this tab to prevent race conditions
    let releaseLock;
    const previousPromise = this.mutexes.get(tabId) || Promise.resolve();
    const nextPromise = new Promise(resolve => { releaseLock = resolve; });
    this.mutexes.set(tabId, previousPromise.then(() => nextPromise));
    await previousPromise;

    try {
      let session = this.sessions.get(tabId);
      if (!session) {
        session = { attached: false, refCount: 0 };
        this.sessions.set(tabId, session);
      }

      if (!session.attached) {
        try {
          await chrome.debugger.attach({ tabId }, '1.3');
          // Enforce target auto-attach to capture popups / OAuth windows
          await chrome.debugger.sendCommand({ tabId }, 'Target.setAutoAttach', {
            autoAttach: true,
            waitForDebuggerOnStart: false,
            flatten: true
          }).catch(() => {}); // Optional, don't fail if Target domain fails
          session.attached = true;
        } catch (e) {
          const msg = e?.message || String(e);
          if (!msg.includes('already attached') && !msg.includes('Another debugger')) {
            releaseLock();
            throw e;
          }
          session.attached = true;
        }
      }

      session.refCount++;

      const send = (method, params = {}) => {
        return new Promise((resolve, reject) => {
          chrome.debugger.sendCommand({ tabId }, method, params, result => {
            if (chrome.runtime.lastError) {
              reject(new Error(chrome.runtime.lastError.message));
            } else {
              resolve(result);
            }
          });
        });
      };

      const release = () => {
        session.refCount = Math.max(0, session.refCount - 1);
        if (session.refCount === 0) {
          // Keep attached for 15 seconds to allow quick follow-up commands, then auto-detach
          setTimeout(async () => {
            if (session.refCount === 0 && session.attached) {
              try {
                await chrome.debugger.detach({ tabId });
              } catch (_) {}
              session.attached = false;
              this.sessions.delete(tabId);
            }
          }, 15000);
        }
        releaseLock();
      };

      return { send, release };
    } catch (err) {
      releaseLock();
      throw err;
    }
  }
}

chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === 'agentExecute') {
    const { tabId, executeAction, params } = request;
    if (!tabId) {
      sendResponse({ success: false, error: 'No active tab target designated.' });
      return true;
    }

    // Wrap the entire action in an immediate async IIFE to prevent service worker timeouts
    (async () => {
      try {
        const result = await handleAgentAction(tabId, executeAction, params);
        sendResponse(result);
      } catch (err) {
        console.error(`[Aion] Agent action "${executeAction}" failed:`, err);
        sendResponse({ success: false, error: err.message || 'Execution error' });
      }
    })();
    return true; // Keep message channel open for async response
  }
  
  // Hand off non-agent actions gracefully...
  return false;
});

async function handleAgentAction(tabId, action, params) {
  // Enforce an absolute 25-second execution timeout to prevent the agent from freezing
  const timeoutPromise = new Promise((_, reject) => 
    setTimeout(() => reject(new Error('Action execution timed out')), 25000)
  );

  const actionPromise = (async () => {
    switch (action) {
      case 'navigate': {
        let url = String(params?.url || '').trim();
        if (!url) throw new Error('No target URL provided');
        if (/^(chrome|chrome-extension|edge|about|devtools):/i.test(url)) {
          throw new Error('Privileged browser URLs are restricted.');
        }
        if (!/^https?:\/\//i.test(url)) url = 'https://' + url;
        
        await chrome.tabs.update(tabId, { url });
        await waitForTabToComplete(tabId);
        return { success: true };
      }

      case 'click': {
        return await executeTrustedClick(tabId, params);
      }

      case 'type': {
        return await executeTrustedType(tabId, params);
      }

      case 'hover': {
        return await executeTrustedHover(tabId, params);
      }

      case 'scroll': {
        const direction = params?.direction || 'down';
        await chrome.scripting.executeScript({
          target: { tabId },
          func: (dir) => {
            const h = window.innerHeight;
            if (dir === 'down') window.scrollBy({ top: h * 0.75, behavior: 'smooth' });
            else if (dir === 'up') window.scrollBy({ top: -h * 0.75, behavior: 'smooth' });
            else if (dir === 'top') window.scrollTo({ top: 0, behavior: 'smooth' });
            else if (dir === 'bottom') window.scrollTo({ top: document.body.scrollHeight, behavior: 'smooth' });
          },
          args: [direction]
        });
        return { success: true };
      }

      case 'waitForElement': {
        const selector = params?.selector;
        const text = params?.text;
        const maxWait = Math.min(20, params?.timeout || 10);
        
        const success = await pollForElement(tabId, selector, text, maxWait);
        if (!success) {
          throw new Error(`Timeout waiting for element ${selector || text}`);
        }
        return { success: true };
      }

      case 'readText': {
        const results = await chrome.scripting.executeScript({
          target: { tabId },
          func: (sel, label) => {
            let el = sel ? document.querySelector(sel) : null;
            if (!el && label) {
              el = Array.from(document.querySelectorAll('*')).find(e => 
                e.textContent.trim().toLowerCase().includes(label.toLowerCase())
              );
            }
            if (!el) return null;
            return el.value || el.innerText || el.textContent || '';
          },
          args: [params?.selector, params?.text]
        });
        const text = results?.[0]?.result;
        if (text === null) throw new Error('Target element not found');
        return { success: true, data: text.trim().slice(0, 3000) };
      }

      default:
        throw new Error(`Action "${action}" is not supported or was deprecated.`);
    }
  })();

  return Promise.race([actionPromise, timeoutPromise]);
}

async function executeTrustedClick(tabId, params) {
  // 1. Try DOM lookup first (Very fast, respects viewport coordinates)
  let coords = await locateElementViaScript(tabId, params);
  
  // 2. Fallback to Accessibility Tree if DOM search fails
  if (!coords && params.text) {
    coords = await locateViaAccessibilityTree(tabId, params.text);
  }

  // 3. Fallback to absolute coordinates if provided
  if (!coords && typeof params.x === 'number' && typeof params.y === 'number') {
    coords = { x: params.x, y: params.y };
  }

  if (!coords) {
    throw new Error(`Could not find click target "${params.text || params.selector || 'element'}"`);
  }

  const session = await cdpPool.acquire(tabId);
  try {
    const jx = Math.round(coords.x + (Math.random() * 4 - 2)); // 2px Jitter to bypass bot detection
    const jy = Math.round(coords.y + (Math.random() * 4 - 2));

    await session.send('Input.dispatchMouseEvent', { type: 'mousePressed', x: jx, y: jy, button: 'left', clickCount: 1 });
    await new Promise(r => setTimeout(r, 60)); // Small, reliable physical delay
    await session.send('Input.dispatchMouseEvent', { type: 'mouseReleased', x: jx, y: jy, button: 'left', clickCount: 1 });
    return { success: true };
  } finally {
    session.release();
  }
}

async function executeTrustedType(tabId, params) {
  let coords = await locateElementViaScript(tabId, params);
  if (!coords && params.selector) {
    coords = { x: 400, y: 400 }; // Fallback center click if coordinates are absent
  }

  const session = await cdpPool.acquire(tabId);
  try {
    // 1. Focus the input box physically
    if (coords) {
      await session.send('Input.dispatchMouseEvent', { type: 'mousePressed', x: coords.x, y: coords.y, button: 'left', clickCount: 1 });
      await session.send('Input.dispatchMouseEvent', { type: 'mouseReleased', x: coords.x, y: coords.y, button: 'left', clickCount: 1 });
      await new Promise(r => setTimeout(r, 100));
    }

    // 2. Clear existing text if requested
    if (params.clearFirst) {
      await session.send('Input.dispatchKeyEvent', { type: 'rawKeyDown', key: 'a', code: 'KeyA', windowsVirtualKeyCode: 65, modifiers: 2 });
      await session.send('Input.dispatchKeyEvent', { type: 'keyUp', key: 'a', code: 'KeyA', windowsVirtualKeyCode: 65, modifiers: 2 });
      await session.send('Input.dispatchKeyEvent', { type: 'rawKeyDown', key: 'Delete', code: 'Delete', windowsVirtualKeyCode: 46 });
      await session.send('Input.dispatchKeyEvent', { type: 'keyUp', key: 'Delete', code: 'Delete', windowsVirtualKeyCode: 46 });
      await new Promise(r => setTimeout(r, 50));
    }

    // 3. Inject text safely
    const text = String(params.text || '');
    await session.send('Input.insertText', { text });

    // 4. Optionally submit the form
    if (params.pressEnter) {
      await session.send('Input.dispatchKeyEvent', { type: 'rawKeyDown', key: 'Enter', code: 'Enter', windowsVirtualKeyCode: 13 });
      await session.send('Input.dispatchKeyEvent', { type: 'keyUp', key: 'Enter', code: 'Enter', windowsVirtualKeyCode: 13 });
    }

    return { success: true };
  } finally {
    session.release();
  }
}

async function executeTrustedHover(tabId, params) {
  const coords = await locateElementViaScript(tabId, params);
  if (!coords) throw new Error('Hover target element not found');

  const session = await cdpPool.acquire(tabId);
  try {
    await session.send('Input.dispatchMouseEvent', { type: 'mouseMoved', x: coords.x, y: coords.y, button: 'none' });
    await new Promise(r => setTimeout(r, 200)); // Allow hover animations to render
    return { success: true };
  } finally {
    session.release();
  }
}

async function locateElementViaScript(tabId, params) {
  try {
    const results = await chrome.scripting.executeScript({
      target: { tabId },
      func: (sel, txt) => {
        let el = sel ? document.querySelector(sel) : null;
        if (!el && txt) {
          el = Array.from(document.querySelectorAll('button, a, input, [role="button"]')).find(e => 
            e.textContent.trim().toLowerCase().includes(txt.toLowerCase())
          );
        }
        if (!el) return null;
        el.scrollIntoView({ block: 'center', inline: 'center' });
        const rect = el.getBoundingClientRect();
        return {
          x: Math.round(rect.left + rect.width / 2),
          y: Math.round(rect.top + rect.height / 2)
        };
      },
      args: [params?.selector, params?.text]
    });
    return results?.[0]?.result || null;
  } catch (_) {
    return null;
  }
}

async function locateViaAccessibilityTree(tabId, text) {
  const session = await cdpPool.acquire(tabId);
  try {
    await session.send('Accessibility.enable').catch(() => {});
    const tree = await session.send('Accessibility.queryAXTree', { accessibleName: text }).catch(() => null);
    if (tree?.nodes?.length > 0) {
      const node = tree.nodes[0];
      if (node.backendDOMNodeId) {
        const box = await session.send('DOM.getBoxModel', { backendNodeId: node.backendDOMNodeId }).catch(() => null);
        if (box?.model?.content) {
          const [x1, y1, x2, , , y3] = box.model.content;
          return { x: (x1 + x2) / 2, y: (y1 + y3) / 2 };
        }
      }
    }
    return null;
  } finally {
    session.release();
  }
}

function waitForTabToComplete(tabId) {
  return new Promise(resolve => {
    const listener = (id, changeInfo) => {
      if (id === tabId && changeInfo.status === 'complete') {
        chrome.tabs.onUpdated.removeListener(listener);
        resolve();
      }
    };
    chrome.tabs.onUpdated.addListener(listener);
    // Timeout safeguard: continue anyway after 8 seconds
    setTimeout(() => {
      chrome.tabs.onUpdated.removeListener(listener);
      resolve();
    }, 8000);
  });
}

function pollForElement(tabId, selector, text, maxSeconds) {
  const end = Date.now() + (maxSeconds * 1000);
  return new Promise(async resolve => {
    while (Date.now() < end) {
      try {
        const results = await chrome.scripting.executeScript({
          target: { tabId },
          func: (sel, txt) => {
            if (sel && document.querySelector(sel)) return true;
            if (txt && document.body.innerText.includes(txt)) return true;
            return false;
          },
          args: [selector, text]
        });
        if (results?.[0]?.result) {
          resolve(true);
          return;
        }
      } catch (_) {}
      await new Promise(r => setTimeout(r, 500)); // Poll every 500ms
    }
    resolve(false);
  });
}

  const apiKey = keyResult.geminiApiKey;
  if (!apiKey) {
    showGeminiModal();
    return;
  }

  // 1. Reset state cleanly
  agentStopRequested = false;
  await chrome.storage.session.remove('agentStopRequested').catch(() => {});
  resetAgentStepLog();

  // 2. Setup Stop Button
  const stopBtn = document.createElement('button');
  stopBtn.textContent = '■ Stop Autopilot';
  stopBtn.className = 'agent-stop-btn';
  stopBtn.style.cssText = 'display:block;margin:8px auto;padding:6px 14px;font-size:12px;border-radius:20px;background:rgba(255,80,80,0.12);border:1px solid rgba(255,80,80,0.3);color:#ff8080;cursor:pointer;';
  stopBtn.addEventListener('click', () => {
    agentStopRequested = true;
    chrome.storage.session.set({ agentStopRequested: true }).catch(() => {});
    stopBtn.disabled = true;
    stopBtn.textContent = 'Stopping...';
  });
  thread.appendChild(stopBtn);

  // 3. Compact UI Transition
  enterAutopilotMiniMode();
  thread.scrollTop = thread.scrollHeight;

  // 4. Resolve Active Tab
  let tab = null;
  try {
    const normalWindows = await chrome.windows.getAll({ populate: true, windowTypes: ['normal'] });
    const candidates = normalWindows
      .flatMap(w => (w.tabs || []).filter(t => t.active).map(t => ({ ...t, _focused: w.focused })))
      .sort((a, b) => (b._focused ? 1 : 0) - (a._focused ? 1 : 0) || (b.lastAccessed || 0) - (a.lastAccessed || 0));
    tab = candidates[0];
  } catch (_) {}

  if (!tab || !tab.id) {
    stopBtn.remove();
    exitAutopilotMiniMode();
    addAgentStepBubble(thread, "Could not locate an active browser tab.", 'error');
    return;
  }

  let lockedTabId = tab.id;
  const contents = [];
  
  // 5. Build Initial Context
  let pageText = await getActiveTabPageText(lockedTabId).catch(() => '');

    role: 'user',
    parts: [{
      text: `TASK: ${prompt}\n\nCURRENT PAGE (${tab.url}):\n${pageText || '(No readable text or blank page)'}`
    }]
  });

  let consecutiveFailures = 0;

  for (let step = 0; step < AGENT_MAX_STEPS; step++) {

      break;
    }

    // Process parallel or single executable tool calls
    const responseParts = [];
    let stepDescription = '';

    for (const part of fnCalls) {
      const { name, args } = part.functionCall;
      stepDescription = `${name}${args?.text ? `: "${args.text}"` : args?.url ? `: ${args.url}` : ''}`;
      addAgentStepBubble(thread, stepDescription);

      const execResult = await chrome.runtime.sendMessage({
        action: 'agentExecute',
        tabId: lockedTabId,
        executeAction: name,
        params: args || {}
      }).catch(e => ({ success: false, error: e.message }));

      if (execResult?.success) {
        consecutiveFailures = 0;
      } else {
        anyFailed = true;
        consecutiveFailures++;
      }

      responseParts.push({
        functionResponse: {
          name,
          response: execResult?.success 
            ? { success: true, data: String(execResult?.data || 'Success') }
            : { success: false, error: execResult?.error || 'Action failed' }
        }
      });
    }

    if (consecutiveFailures >= 5) {
      addAgentStepBubble(thread, '⛔ Too many consecutive failures. Safely stopping.', 'error');
      break;
    }

    // Wait for the viewport to settle, and append results for the next turn
    await new Promise(r => setTimeout(r, 600));
    pageText = await getActiveTabPageText(lockedTabId).catch(() => '');
    
    // Inject current page state into every response part for model alignment
    responseParts.forEach(rp => {
      rp.functionResponse.response.pageNow = pageText.slice(0, 3000);
    });

    contents.push({ role: 'user', parts: responseParts });
  }

  stopBtn.remove();
  exitAutopilotMiniMode();
}
