// NEW ULTRA-FAST ANNOTATION EDITOR — REPLACE EVERYTHING
function openAnnotationEditor(imageDataUrl) {
  const win = window.open('', 'SnapToAI Editor', 'width=1200,height=800');
  win.document.write(`
    <!DOCTYPE html>
    <html>
    <head>
      <title>SnapToAI Editor</title>
      <style>
        body,html{margin:0;padding:0;background:#111;color:#fff;font-family:system-ui}
        canvas{cursor:crosshair;display:block;margin:20px auto;border:2px solid #007AFF}
        .tools{position:fixed;top:10px;left:50%;transform:translateX(-50%);background:#000;padding:10px;border-radius:12px;z-index:10}
        button{margin:0 5px;padding:10px 20px;background:#007AFF;color:white;border:none;border-radius:8px;cursor:pointer}
        button.active{background:#00d9ff}
      </style>
    </head>
    <body>
      <div class="tools">
        <button id="pen" class="active">Pen</button>
        <button id="highlighter">Highlighter</button>
        <button id="text">Text</button>
        <button id="undo">Undo</button>
        <button id="save">SAVE & COPY</button>
      </div>
      <canvas id="c"></canvas>
      <script>
        const canvas = document.getElementById('c');
        const ctx = canvas.getContext('2d', { willReadFrequently: true });
        let drawing = false, tool = 'pen', color = '#007AFF', size = 4;
        const history = [];
        
        const img = new Image();
        img.onload = () => {
          canvas.width = img.width;
          canvas.height = img.height;
          ctx.drawImage(img, 0, 0);
          saveState();
        };
        img.src = "${imageDataUrl}";
        
        function saveState() {
          history.push(canvas.toDataURL());
        }
        
        canvas.addEventListener('mousedown', e => {
          drawing = true;
          ctx.beginPath();
          ctx.moveTo(e.offsetX, e.offsetY);
        });
        canvas.addEventListener('mousemove', e => {
          if (!drawing) return;
          ctx.globalCompositeOperation = tool === 'highlighter' ? 'multiply' : 'source-over';
          ctx.strokeStyle = tool === 'highlighter' ? 'rgba(255,255,0,0.5)' : color;
          ctx.lineWidth = tool === 'highlighter' ? 20 : size;
          ctx.lineCap = 'round';
          ctx.lineTo(e.offsetX, e.offsetY);
          ctx.stroke();
        });
        canvas.addEventListener('mouseup', () => {
          drawing = false;
          saveState();
        });
        
        document.getElementById('pen').onclick = () => tool = 'pen';
        document.getElementById('highlighter').onclick = () => tool = 'highlighter';
        document.getElementById('undo').onclick = () => {
          if (history.length > 1) {
            history.pop();
            const img = new Image();
            img.onload = () => ctx.drawImage(img, 0, 0);
            img.src = history[history.length - 1];
          }
        };
        document.getElementById('save').onclick = () => {
          canvas.toBlob(blob => {
            navigator.clipboard.write([new ClipboardItem({'image/png': blob})]);
            alert('Saved & copied to clipboard!');
          });
        };
      </script>
    </body>
    </html>
  `);
}
