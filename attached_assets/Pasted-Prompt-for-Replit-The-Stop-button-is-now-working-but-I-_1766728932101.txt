function speakText(text, langCode = 'en-US') {
    // Stop previous audio
    synth.cancel(); 

    const utterance = new SpeechSynthesisUtterance(text);
    
    // Refresh voices list in case they just loaded
    if (voices.length === 0) voices = synth.getVoices();

    // 1. Try to find a high-quality Google/Natural voice for the specific language
    // 2. Fallback to any voice for that language
    // 3. Fallback to any English voice if all else fails
    const bestVoice = voices.find(v => v.lang.startsWith(langCode) && (v.name.includes('Google') || v.name.includes('Natural'))) ||
                      voices.find(v => v.lang.startsWith(langCode)) ||
                      voices.find(v => v.lang.startsWith('en'));

    if (bestVoice) {
        utterance.voice = bestVoice;
        utterance.lang = bestVoice.lang;
        console.log(`[SnapToAI] Using voice: ${bestVoice.name} for language: ${langCode}`);
    } else {
        utterance.lang = langCode;
    }

    utterance.rate = 1.0;
    utterance.pitch = 1.0;
    
    synth.speak(utterance);
}

        speakText(request.text, request.lang || 'en-US');
        sendResponse({ success: true });
