Step 1: The "Flashcard" Prompt (logic.js)
We need to tell Gemini to stop talking like a human and start talking like a computer (JSON) so we can build the cards programmatically.

Add this function to your logic.js:

JavaScript

async function generateFlashcards(topic) {
  const prompt = `Create 5 study flashcards about "${topic}". 
  Strictly output ONLY a JSON array with this format: 
  [{"front": "Question/Term", "back": "Answer/Definition"}]
  Do not write any other text.`;

  const response = await callGeminiAPI(prompt); // Use your existing API caller
  
  // Clean the response (sometimes AI adds markdown ```json blocks)
  const cleanJson = response.replace(/```json|```/g, '').trim();
  const cards = JSON.parse(cleanJson);
  
  renderFlashcards(cards);
}

🛠️ Step 3: The Builder & PDF Export (logic.js)
This code takes the JSON from Gemini and builds the HTML cards, plus adds a "Print to PDF" feature.

JavaScript

function renderFlashcards(cards) {
  const container = document.getElementById('chat-box'); // Or a new div 'flashcard-area'
  container.innerHTML = '<div class="flashcard-grid" id="printable-area"></div>';
  const grid = container.querySelector('.flashcard-grid');

  cards.forEach(card => {
    const cardHTML = `
      <div class="flashcard" onclick="this.classList.toggle('flipped')">
        <div class="flashcard-inner">
          <div class="flashcard-front">${card.front}</div>
          <div class="flashcard-back">${card.back}</div>
        </div>
      </div>
    `;
    grid.innerHTML += cardHTML;
  });

  // Add the Download Button
  const btn = document.createElement('button');
  btn.innerText = "📄 Save as PDF";
  btn.className = "download-btn"; // Style this to look nice!
  btn.onclick = () => printFlashcards();
  container.appendChild(btn);
}

function printFlashcards() {
  // Simple "Print" trigger - user selects "Save as PDF" in the print dialog
  const printContent = document.getElementById('printable-area').innerHTML;
  const win = window.open('', '', 'height=600,width=800');
  win.document.write('<html><head><title>My Flashcards</title>');
  // Add basic styles for the PDF
  win.document.write('<style>.flashcard { border: 1px solid #333; margin: 10px; padding: 20px; page-break-inside: avoid; }</style>'); 
  win.document.write('</head><body>');
  win.document.write(printContent);
  win.document.write('</body></html>');
  win.document.close();
  win.print();
}
🌟 Why this is a "Wow" Feature
Visual Interaction: Users aren't just reading text; they are playing with the cards (flipping them).

Tangible Value: The ability to "Save as PDF" means they can print them out and cut them with scissors for real-world studying.

Structured Learning: It forces Gemini to organize messy information into clean key points.
