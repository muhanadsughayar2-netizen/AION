2. The Combined Logic (JavaScript)
This handles multiple files, creates the "Cards," and prepares the Gemini API payload all in one go.

JavaScript

let filesQueue = []; // This replaces your single global variable

// 1. Handle File Selection & Card Creation
document.getElementById('file-input').onchange = (e) => {
  Array.from(e.target.files).forEach(file => {
    const reader = new FileReader();
    reader.onload = (event) => {
      const fileData = { mime_type: file.type, data: event.target.result.split(',')[1], name: file.name };
      filesQueue.push(fileData);
      
      // Add the "Card" to UI
      const card = document.createElement('div');
      card.className = 'file-card';
      card.innerHTML = `📄 <span>${file.name}</span> <div class="remove-btn">×</div>`;
      card.querySelector('.remove-btn').onclick = () => {
        filesQueue = filesQueue.filter(f => f !== fileData);
        card.remove();
      };
      document.getElementById('file-preview-zone').appendChild(card);
    };
    reader.readAsDataURL(file);
  });
};

// 2. The Professional Send Function
async function sendMessage() {
  const text = document.getElementById('chat-input').value;
  const payload = { contents: [{ parts: [{ text: text }] }] };

  // Attach all files from the queue to the Gemini payload
  filesQueue.forEach(f => payload.contents[0].parts.push({ inline_data: { mime_type: f.mime_type, data: f.data } }));

  const response = await fetch(YOUR_API_URL, { method: 'POST', body: JSON.stringify(payload) });
  
  // Clean up
  filesQueue = [];
  document.getElementById('file-preview-zone').innerHTML = '';
  document.getElementById('chat-input').value = '';
}
What this achieves:
One Button (+): Replaces the paperclip with a clean plus sign.

Gemini Box UI: Files appear as gray rounded "Cards" with a logo and an (x) to delete them.

Multi-File: You can upload a PDF and an Image at the same time, and Gemini will "see" both.

Efficient Payload: Everything (text + all files) is sent in a single contents array as required by the Gemini 3 Flash API.
