async function startAnalysis(prompt, base64Image) {
  try {
    const result = await model.generateContent([prompt, { inlineData: { data: base64Image, mimeType: "image/jpeg" } }]);
    const response = await result.response;
    
    // 1. Get the Audio Data
    const audioData = response.candidates[0].content.parts.find(p => p.inlineData)?.inlineData.data;

    if (audioData) {
      // 2. Play the Native Gemini 3 Voice
      const audio = new Audio(`data:audio/wav;base64,${audioData}`);
      audio.play();
      console.log("Playing Native Gemini 3.0 Audio...");
    } else {
      // 3. Emergency Fallback (If API is busy)
      const text = response.text();
      const utterance = new SpeechSynthesisUtterance(text);
      window.speechSynthesis.speak(utterance);
    }
  } catch (err) {
    console.error("Error calling Gemini 3 Flash:", err);
  }
}
