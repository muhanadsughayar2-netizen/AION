// SNAPTOAI MOUSE WAND MENU (Right-click controls everything)
// Reuses ALL your existing functions - no other files touched
// ============================================
function registerSnapToAIMenu() {
  chrome.contextMenus.removeAll(() => {
    chrome.contextMenus.create({
      id: "snaptoai-parent",
      title: "📸 SnapToAI ✨",
      contexts: ["all"]
    });

    // Free actions
    chrome.contextMenus.create({ id: "snap-viewport", title: "Snap Viewport", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "snip-region", title: "Snip Region", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "full-page", title: "Full Page Capture", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "add-to-story", title: "Add to Story", parentId: "snaptoai-parent", contexts: ["all"] });

    chrome.contextMenus.create({ id: "sep1", type: "separator", parentId: "snaptoai-parent" });

    // AI actions (opens your beautiful Gemini chat)
    chrome.contextMenus.create({ id: "ask-ai-this", title: "Ask AI About This", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "explain-text", title: "Explain Selected Text", parentId: "snaptoai-parent", contexts: ["selection"] });
    chrome.contextMenus.create({ id: "analyze-image", title: "Analyze This Image", parentId: "snaptoai-parent", contexts: ["image"] });

    chrome.contextMenus.create({ id: "sep2", type: "separator", parentId: "snaptoai-parent" });

    // Pro features (users see them and want to pay)
    chrome.contextMenus.create({ id: "full-story-ai", title: "💎 Send Full Story to AI", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "agent-auto", title: "💎 Trigger Agent Automation", parentId: "snaptoai-parent", contexts: ["all"] });
    chrome.contextMenus.create({ id: "view-queue", title: "View Queue", parentId: "snaptoai-parent", contexts: ["all"] });
  });
}
registerSnapToAIMenu();

chrome.contextMenus.onClicked.addListener(async (info, tab) => {
  if (!tab?.id) return;

  switch (info.menuItemId) {
    case "snap-viewport":
      captureScreenshot();                    // your existing function
      break;
    case "snip-region":
      chrome.tabs.sendMessage(tab.id, { action: "startSnip" });  // your existing snip flow
      break;
    case "full-page":
      startFullPageCapture();                 // your existing function
      break;
    case "add-to-story":
      captureScreenshot();                    // adds to queue
      break;
    case "ask-ai-this":
    case "explain-text":
    case "analyze-image":
      handleAskSnapToAI(info, tab);           // your existing AI chat (opens your Gemini window)
      break;
    case "full-story-ai":
      // Opens your Gemini chat with full queue (reuses your AI chat code)
      chrome.tabs.sendMessage(tab.id, { action: "captureAndOpenAIChat", from: "mouse", fullQueue: true });
      break;
    case "agent-auto":
      chrome.tabs.sendMessage(tab.id, { action: "triggerAgent" });
      break;
    case "view-queue":
      chrome.action.openPopup();              // your existing popup
      break;
  }
});
