      window.calculateBMI = function() {
        const w = parseFloat(document.getElementById('bmi-weight').value);
        let h = parseFloat(document.getElementById('bmi-height').value);
        if (!w || !h) return;

        let bmi;
        if (appState.bmiUnit === 'metric') {
          bmi = w / ((h/100) * (h/100));
        } else {
          bmi = (w / (h * h)) * 703;
        }

        const resCard = document.getElementById('bmi-result-card');
        const valEl = document.getElementById('bmi-value');
        const catEl = document.getElementById('bmi-category');
        const barEl = document.getElementById('bmi-bar');
        const adviceEl = document.getElementById('bmi-advice');

        resCard.classList.remove('hidden');
        valEl.textContent = bmi.toFixed(1);

        let cat, color, advice, perc;
        if (bmi < 18.5) { cat = 'Underweight'; color = '#3b82f6'; perc = 25; advice = "Time to fuel up! Focus on nutrient-dense meals and strength training."; }
        else if (bmi < 25) { cat = 'Normal'; color = '#00FF88'; perc = 50; advice = "Perfect balance! Keep doing what you're doing."; }
        else if (bmi < 30) { cat = 'Overweight'; color = '#fbbf24'; perc = 75; advice = "Let's kick it up a notch. Try adding more HIIT to your routine."; }
        else { cat = 'Obese'; color = '#ef4444'; perc = 100; advice = "Don't discourage! Consistency is key. Our trainers can help you build a safe plan."; }

        catEl.textContent = cat;
        catEl.style.color = color;
        barEl.style.backgroundColor = color;
        barEl.style.width = perc + '%';
        adviceEl.textContent = advice;
      };

      // 5. Visual Effects & Utils
      function initScrollEffects() {
        const nav = document.getElementById('main-nav');
        window.addEventListener('scroll', () => {
          nav.classList.toggle('nav-scrolled', window.scrollY > 50);
        });

        const io = new IntersectionObserver(entries => {
          entries.forEach(e => {
            if (e.isIntersecting) {
              e.target.classList.add('visible');
              io.unobserve(e.target);
            }
          });
        }, { threshold: 0.1 });
        document.querySelectorAll('.reveal').forEach(el => io.observe(el));
      }

      function initCounters() {
        const counters = document.querySelectorAll('.counter');
        const io = new IntersectionObserver(entries => {
          entries.forEach(e => {
            if (e.isIntersecting) {
              const target = +e.target.dataset.target;
              let current = 0;
              const step = target / 50;
              const timer = setInterval(() => {
                current += step;
                if (current >= target) {
                  e.target.textContent = target.toLocaleString() + '+';
                  clearInterval(timer);
                } else {
                  e.target.textContent = Math.floor(current).toLocaleString() + '+';
                }
              }, 30);
              io.unobserve(e.target);
            }
          });
        });
        counters.forEach(c => io.observe(c));
      }

      function initCountdown() {
        const el = document.getElementById('countdown');
        let seconds = 3600 * 2.5; // 2.5 hours
        setInterval(() => {
          seconds--;
          const h = Math.floor(seconds / 3600);
          const m = Math.floor((seconds % 3600) / 60);
          const s = seconds % 60;
          el.innerHTML = `<span>${String(h).padStart(2,'0')}</span>:<span>${String(m).padStart(2,'0')}</span>:<span>${String(s).padStart(2,'0')}</span>`;
        }, 1000);
      }

      function initBeforeAfter() {
        const container = document.getElementById('ba-container');
        const after = document.getElementById('ba-after');
        const handle = document.getElementById('ba-handle');
        
        container.addEventListener('mousemove', (e) => {
          const rect = container.getBoundingClientRect();
          const x = e.clientX - rect.left;
          const perc = (x / rect.width) * 100;
          after.style.width = perc + '%';
          handle.style.left = perc + '%';
        });
      }

      function spawnConfetti() {
        for(let i=0; i<100; i++) {
          const c = document.createElement('div');
          c.className = 'confetti';
          c.style.left = Math.random() * 100 + 'vw';
          c.style.top = '-10px';
          c.style.backgroundColor = Math.random() > 0.5 ? '#00FF88' : '#fff';
          c.style.transform = `rotate(${Math.random()*360}deg)`;
          document.body.appendChild(c);
          
          const destX = (Math.random() - 0.5) * 500;
          const destY = window.innerHeight + 100;
          
          c.animate([
            { transform: `translate(0, 0) rotate(0deg)`, opacity: 1 },
            { transform: `translate(${destX}px, ${destY}px) rotate(720deg)`, opacity: 0 }
          ], { duration: 2000 + Math.random()*1000, easing: 'cubic-bezier(0,0,0.2,1)' }).onfinish = () => c.remove();
        }
      }
