    // 1. Reveal Elements on Scroll
    const _io = new IntersectionObserver(entries => {
      entries.forEach(e => {
        if (e.isIntersecting) {
          e.target.classList.add('visible');
          _io.unobserve(e.target);
        }
      });
    }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

    document.querySelectorAll('.reveal').forEach(el => _io.observe(el));

    // 2. Tab Menu Toggling
    function showTab(id, btn) {
      document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      
      const targetPanel = document.getElementById(id);
      if (targetPanel) targetPanel.classList.add('active');
      if (btn) btn.classList.add('active');
    }

    // 3. Metabolic Cumulative Selector Logic
    let runningScore = 0;
    const textPercent = document.getElementById('percentText');
    const visualCircle = document.getElementById('progressCircle');
    const activeLabel = document.getElementById('activeFoodName');
    const circleCircumference = 2 * Math.PI * 130; // 816.814 px

    document.querySelectorAll('.bio-food-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        const value = parseInt(btn.getAttribute('data-value'));
        const color = btn.getAttribute('data-color');
        const name = btn.getAttribute('data-name');
        
        if (btn.classList.contains('active')) {
          btn.classList.remove('active');
          runningScore -= value;
          activeLabel.innerText = "Deactivated bioactive path.";
          activeLabel.style.color = "var(--text-muted)";
        } else {
          btn.classList.add('active');
          runningScore += value;
          activeLabel.innerText = "Activated: " + name;
          activeLabel.style.color = color;
        }

        updateDashboard(runningScore, color);
      });
    });

    function updateDashboard(score, colorCode) {
      const displayScore = Math.min(Math.max(score, 0), 100);
      
      // Compute progress ring stroke-dashoffset
      const targetOffset = circleCircumference - (displayScore / 100) * circleCircumference;
      visualCircle.style.stroke = score > 0 ? colorCode : 'var(--accent)';
      visualCircle.style.strokeDashoffset = targetOffset;

      // Animate percentage label counter
      const currentNumber = parseInt(textPercent.textContent);
      const duration = 500;
      const startTime = performance.now();

      function stepAnimation(now) {
        const elapsed = now - startTime;
        const progress = Math.min(elapsed / duration, 1);
        const interpolatedValue = Math.floor(currentNumber + (displayScore - currentNumber) * progress);
        textPercent.innerText = interpolatedValue + '%';

        if (progress < 1) requestAnimationFrame(stepAnimation);
      }
      requestAnimationFrame(stepAnimation);
    }

    // 4. Smooth Scrolling with Offset Target Logic
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
      anchor.addEventListener('click', function (e) {
        e.preventDefault();
        const targetId = this.getAttribute('href');
        const targetElement = document.querySelector(targetId);
        
        if (targetElement) {
          const headerOffset = 90; // Fixed navbar height offset guard
          const elementPosition = targetElement.getBoundingClientRect().top;
          const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
  
          window.scrollTo({
            top: offsetPosition,
            behavior: "smooth"
          });
        }
      });
    });

    // 5. Connect Dead Pricing/CTAs to Scroll directly to Visualizer Tool
