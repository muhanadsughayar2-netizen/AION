2. Content Script: "Mutation Storm" Prevention
The Problem: In content.js, your grokObserver and inputProtectionObserver observe the entire document.body with subtree: true. On highly dynamic sites (like Twitter/X or Gmail), this can trigger hundreds of times per second, causing "jank" (input lag) for the user.

The Fix: Use a Debounce function on your observers.

// Add this helper to the top of content.js
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}

async function waitForImages() {
  const images = Array.from(document.querySelectorAll('img'));
  await Promise.all(images.map(img => {
