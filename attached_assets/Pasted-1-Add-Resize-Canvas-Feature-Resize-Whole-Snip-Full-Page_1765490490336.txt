      tempCtx.drawImage(canvas, 0, 0, newWidth, newHeight);
      
      // Resize main canvas

      canvas.height = newHeight;
      ctx.drawImage(tempCanvas, 0, 0);
      
      // Update originalImage
      originalImage = ctx.getImageData(0, 0, canvas.width, canvas.height);
      
      // Scale all annotations
      const scaleX = newWidth / canvas.width; // old width was previous
      const scaleY = newHeight / canvas.height;
      annotations.forEach(ann => {
        ann.x *= scaleX;
        ann.y *= scaleY;
        if (ann.width) ann.width *= scaleX;
        if (ann.height) ann.height *= scaleY;
        if (ann.x1) { ann.x1 *= scaleX; ann.x2 *= scaleX; }
        if (ann.y1) { ann.y1 *= scaleY; ann.y2 *= scaleY; }
      });
      
      redraw();
      updateCssBorder();
      updateBrowserFrameOverlay();
      updateStatus(`Resized to ${newWidth}×${newHeight}`);
    } else {
      alert('Invalid size. Use 100–5000px.');
    }
  }

  const h = 70;

  // Background
  ctx.fillStyle = ann.color || '#FFFF88';
  ctx.shadowColor = 'rgba(0,0,0,0.25)';
  ctx.shadowBlur = 10;
  ctx.shadowOffsetY = 4;
  ctx.fillRect(-w/2, -h/2, w, h);

  // Border
  ctx.strokeStyle = 'rgba(0,0,0,0.2)';
  ctx.lineWidth = 2;
  ctx.strokeRect(-w/2, -h/2, w, h);

  // Text — always visible
  ctx.fillStyle = textColor;
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText(ann.text, 0, 0);

  ctx.restore();
}

// Helper: auto black/white text
function getContrastColor(hex) {
  const r = parseInt(hex.substr(1,2),16);
  const g = parseInt(hex.substr(3,2),16);
  const b = parseInt(hex.substr(5,2),16);
  const yiq = ((r*299)+(g*587)+(b*114))/1000;
  return (yiq >= 128) ? '#000000' : '#FFFFFF';
}
