        // --- GOOGLE DOCS SPECIAL HANDLING ---
        if (url.hostname.includes('docs.google.com') && url.pathname.includes('/document/')) {
          await chrome.scripting.executeScript({
            target: { tabId },
            func: async (textToType) => {
              const iframe = document.querySelector('.docs-texteventtarget-iframe');
              if (iframe && iframe.contentDocument) {
                const body = iframe.contentDocument.body;
                body.focus();
                // Force a small delay for focus to register
                await new Promise(r => setTimeout(r, 100));
                // Use the only command Google Docs' internal listener accepts
                iframe.contentDocument.execCommand('insertText', false, textToType);
                return { success: true, method: 'iframe-insert' };
              }
              return { success: false, error: 'Iframe not found' };
            },
            args: [params.text]

          sendResponse({ success: true });
