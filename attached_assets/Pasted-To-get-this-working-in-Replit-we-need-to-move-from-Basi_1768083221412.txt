// This function pulls the 3 stored images and sends them to Gemini 3 Flash
async function runSuperAnalysis() {
  const data = await chrome.storage.local.get(['snap_income', 'snap_balance', 'snap_cashflow']);
  
  const systemPrompt = `
    You are a Senior Wall Street Equity Analyst Agent. 
    I am providing 3 images: Income Statement, Balance Sheet, and Cash Flow.
    
    TASK:
    1. Extract all numbers from the images using your vision capabilities.
    2. Perform a "Fundamental Analysis Report."
    3. Calculate: P/E Ratio, Debt-to-Equity, and Free Cash Flow.
    4. Provide a "SWOT Analysis" based on the data.
    5. Final Verdict: Strong Buy, Buy, Hold, or Sell.
  `;

  // Send the images + prompt to Gemini
  const response = await callGeminiAPI(systemPrompt, [
    data.snap_income, 
    data.snap_balance, 
    data.snap_cashflow
  ]);

  return response;
}
