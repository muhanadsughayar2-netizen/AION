// The capture function that runs INSIDE the page/iframes
function runGrokProofCapture() {
  try {
    let images = [];
    let currentPos = window.scrollY || 0;
    const viewportHeight = window.innerHeight;
    const totalHeight = Math.max(
      document.body.scrollHeight, document.documentElement.scrollHeight, window.innerHeight
    );

    const captureChunk = async () => {
      // Capture current visible viewport
      const dataUrl = await new Promise(resolve => {
        chrome.runtime.sendMessage({ action: 'captureVisible' }, resolve);
      });
      images.push(dataUrl);

      // Scroll down one viewport (or up if at bottom – Grok starts scrolled)
      const nextPos = Math.min(currentPos + viewportHeight, totalHeight - viewportHeight);
      window.scrollTo(0, nextPos);
      currentPos = nextPos;

      // Wait for render (Grok needs this for lazy-load messages)
      await new Promise(resolve => setTimeout(resolve, 400));

      if (currentPos < totalHeight - viewportHeight) {
        captureChunk();
      } else {
        // Done – send back to background
        chrome.runtime.sendMessage({ action: 'fullPageDone', images });
      }
    };

    captureChunk();
  } catch (e) {
    // Silent fail – no error popup, just log
    console.warn('Capture skipped on this frame (Grok protection) – continuing...');
  }
}
