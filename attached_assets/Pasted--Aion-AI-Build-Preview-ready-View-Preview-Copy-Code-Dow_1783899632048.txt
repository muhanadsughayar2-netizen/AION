  <script>
    (function() {
      const FEATURES = [
        { id: 1, name: 'Vision', type: 'Utility', desc: 'Understand images and visual data in real-time.', icon: 'search', color: '#3B82F6', bg: 'bg-blue-50' },
        { id: 2, name: 'Image', type: 'Creative', desc: 'Generate high-fidelity artwork from text.', icon: 'sparkles', color: '#F59E0B', bg: 'bg-amber-50' },
        { id: 3, name: 'Music', type: 'Creative', desc: 'Compose royalty-free soundtracks instantly.', icon: 'music', color: '#8B5CF6', bg: 'bg-purple-50' },
        { id: 4, name: 'Video', type: 'Creative', desc: 'Create cinematic clips with simple prompts.', icon: 'video', color: '#EF4444', bg: 'bg-red-50' },
        { id: 5, name: 'Broadcast', type: 'Utility', desc: 'Live AI narration and content streaming.', icon: 'mic', color: '#EC4899', bg: 'bg-pink-50' },
        { id: 6, name: 'Autopilot', type: 'Utility', desc: 'Automate repetitive browser tasks securely.', icon: 'compass', color: '#10B981', bg: 'bg-emerald-50' },
        { id: 7, name: 'Custom Prompts', type: 'Utility', desc: 'Save and organize your favorite AI workflows.', icon: 'terminal', color: '#6366F1', bg: 'bg-indigo-50' },
        { id: 8, name: 'Build', type: 'Creative', desc: 'No-code tool to create your own AI agents.', icon: 'hammer', color: '#F97316', bg: 'bg-orange-50' }
      ];

      const state = {
        activeCategory: 'All'
      };

      window.addEventListener('load', () => {
        lucide.createIcons();
        renderFeatures();
        
        const observer = new IntersectionObserver((entries) => {
          entries.forEach(entry => {
            if (entry.isIntersecting) {
              entry.target.classList.add('visible');
              observer.unobserve(entry.target);
            }
          });
        }, { threshold: 0.1 });

        document.querySelectorAll('.reveal').forEach(el => observer.observe(el));
      });

      window.renderFeatures = function() {
        const grid = document.getElementById('feature-grid');
        const filtered = FEATURES.filter(f => state.activeCategory === 'All' || f.type === state.activeCategory);

        grid.innerHTML = filtered.map((f, idx) => `
          <div class="feature-card group bg-white border border-gray-100 p-8 rounded-3xl reveal visible" style="animation-delay: ${idx * 0.05}s">
            <div class="w-14 h-14 ${f.bg} rounded-2xl flex items-center justify-center mb-6 group-hover:scale-110 transition-transform duration-300">
              <i data-lucide="${f.icon}" style="color: ${f.color}" class="w-7 h-7"></i>
            </div>
            <div class="flex items-center gap-2 mb-3">
              <h3 class="font-bold text-xl text-gray-900">${f.name}</h3>
              <span class="px-2 py-0.5 bg-gray-50 text-[10px] text-muted font-bold rounded uppercase tracking-wider">${f.type}</span>
            </div>
            <p class="text-muted text-sm leading-relaxed mb-6">${f.desc}</p>
            <button class="text-sm font-bold text-gray-900 flex items-center gap-2 group/btn">
              Explore ${f.name} <i data-lucide="arrow-right" class="w-4 h-4 group-hover/btn:translate-x-1 transition-transform"></i>
            </button>
          </div>
        `).join('');
        lucide.createIcons();
      };

      window.filterFeatures = function(category, btn) {
        state.activeCategory = category;
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        renderFeatures();
      };

      window.showToast = function(msg) {
        const toast = document.getElementById('toast');
        const message = document.getElementById('toast-message');
        message.textContent = msg;
        toast.classList.remove('opacity-0', 'translate-y-4', 'pointer-events-none');
        setTimeout(() => {
          toast.classList.add('opacity-0', 'translate-y-4', 'pointer-events-none');
        }, 3000);
      };

      // Smooth scroll adjustment
      document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
          e.preventDefault();
          const target = document.querySelector(this.getAttribute('href'));
          if (target) {
            window.scrollTo({
              top: target.offsetTop - 90,
              behavior: 'smooth'
            });
          }
        });
      });

    })();
  </script>
