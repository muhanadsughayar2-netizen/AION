1. manifest.json (Key Part)
JSON{
  "manifest_version": 3,
  "name": "SnapToAI",
  "version": "1.0",
  "permissions": ["activeTab"],  // This unlocks sensitive pages!
  "action": {
    "default_popup": "popup.html",
    "default_icon": "icon.png"
  },
  "background": {
    "service_worker": "background.js"
  },
  "content_scripts": [
    {
      "matches": ["<all_urls>"],  // For overlay on any page
      "js": ["content.js"]
    }
  ]
}

3. popup.js (The Activation Trigger)
JavaScriptdocument.getElementById('snip').addEventListener('click', () => {
  // User clicked = activeTab grants temp powers!
  chrome.tabs.query({active: true, currentWindow: true}, (tabs) => {
    const tab = tabs[0];
    chrome.tabs.sendMessage(tab.id, {action: 'startSnip'});  // Tell content script to show overlay
  });
  window.close();  // Close popup
});
4. content.js (The Overlay + Capture Magic)
JavaScriptlet startX, startY, endX, endY;
let overlay = null;

chrome.runtime.onMessage.addListener((msg) => {
  if (msg.action === 'startSnip') {
    createOverlay();
  }
});

function createOverlay() {
  overlay = document.createElement('div');
  overlay.style.cssText = `
    position: fixed; top: 0; left: 0; width: 100vw; height: 100vh;
    background: rgba(0,0,0,0.5); z-index: 999999; cursor: crosshair;
    pointer-events: all;
  `;
  document.body.appendChild(overlay);

  let isDragging = false;
  overlay.addEventListener('mousedown', (e) => {
    isDragging = true;
    startX = e.clientX;
    startY = e.clientY;
    drawRect(startX, startY, 0, 0);
  });

  overlay.addEventListener('mousemove', (e) => {
    if (!isDragging) return;
    endX = e.clientX;
    endY = e.clientY;
    drawRect(startX, startY, endX - startX, endY - startY);
  });

  overlay.addEventListener('mouseup', () => {
    if (isDragging) {
      isDragging = false;
      captureAndCleanup();
    }
  });
}

function drawRect(x, y, w, h) {
  if (overlay) {
    overlay.innerHTML = `<div style="
      position: absolute; top: ${y}px; left: ${x}px; width: ${Math.abs(w)}px; height: ${Math.abs(h)}px;
      background: rgba(0,0,0,0.2); border: 2px dashed #fff; pointer-events: none;
    "></div>`;
  }
}

function cleanup() {
  if (overlay) {
    document.body.removeChild(overlay);
