I have investigated the race condition warning and can confirm your architect was 100% correct about the root cause.
When the separate tab page loads, the browser compiles and runs preview-output.js instantly. It queries chrome.storage.local and fires a postMessage containing the compiled HTML code directly to the child iframe.
However, since sandbox.html has to resolve and download the Tailwind CDN script, compile the visual assets, and set up the browser's DOM structure, it hasn't registered its window.addEventListener('message') listener yet. The message lands in an empty inbox, is completely dropped, and leaving you with a blank black screen.

2. preview-output.js (The Tab Handshake Engine)
Create of replace your current preview-output.js with this handshake script:
code
JavaScript
// preview-output.js
// Handles displaying the compiled website in a separate tab with sandboxed rendering

let isSandboxReady = false;
let queuedCode = null;

// Hide the loading overlay with a subtle fade-out transition
function hideLoader() {
  const loader = document.getElementById('loadingOverlay');
  if (loader) {
    loader.style.opacity = '0';
    setTimeout(() => {
      loader.style.display = 'none';
    }, 300);
  }
}

// 1. Establish the handshake listener
window.addEventListener('message', function(event) {
  if (event.data && event.data.sandboxReady) {
    console.log('[Preview Output] Sandbox handshake received! Sandbox is ready.');
    isSandboxReady = true;
    
    // Deliver the queued code immediately
    if (queuedCode) {
      console.log('[Preview Output] Flushing queued HTML code to sandbox.');
      postToSandbox(queuedCode);
      queuedCode = null;
    }
    hideLoader();
  }
});

// Sends the HTML code into the sandboxed frame securely via postMessage
function postToSandbox(code) {
  const iframe = document.getElementById('livePreview');
  if (!iframe) return;
  
  if (iframe.contentWindow) {
    iframe.contentWindow.postMessage({ htmlCode: code }, '*');
  }
}

// 2. Fetch the compiled HTML from local storage
chrome.storage.local.get('snaptoai_built_code', (res) => {
  const code = res.snaptoai_built_code;
  if (code) {
    console.log('[Preview Output] Retrieved site from storage. Size:', code.length, 'chars');
    if (isSandboxReady) {
      postToSandbox(code);
    } else {
      console.log('[Preview Output] Sandbox not ready yet. Queuing code.');
      queuedCode = code;
    }
  } else {
    console.warn('[Preview Output] No compiled code found in storage.');
    hideLoader();
    document.body.innerHTML = `
      <div style="color: #ff6b6b; display: flex; flex-direction: column; align-items: center; justify-content: center; height: 100vh; font-family: sans-serif; gap: 10px;">
        <span style="font-size: 32px">⚠️</span>
        <strong style="font-size: 16px">No preview available</strong>
        <span style="font-size: 12px; color: #8899aa">Generate or compile a website in the main panel first.</span>
      </div>
    `;
  }
});

// 3. Safety Net — If the sandbox fails to load (CDNs down, network issue etc.)
// ensure the loading spinner is dismissed after 5 seconds to prevent frozen hangs
setTimeout(() => {
  if (!isSandboxReady) {
    console.warn('[Preview Output] Handshake timeout. Forcing loader removal.');
    if (queuedCode) {
      postToSandbox(queuedCode);
    }
    hideLoader();
  }
}, 5000);
3. sandbox.html (The Sandbox Environment)
This file safely captures compiled output payloads, extracts CSS styles, executes the inner javascript, and transmits its initialization state via postMessage:
code
Html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>SnapToAI Sandbox</title>
  <script src="https://cdn.tailwindcss.com"></script>
  <script src="https://unpkg.com/lucide@latest"></script>
  <script>
    try {
      tailwind.config = {
        theme: {
          extend: {
            fontFamily: {
              sans: ['"Plus Jakarta Sans"', 'system-ui', 'sans-serif'],
              serif: ['"Playfair Display"', 'serif'],
              mono: ['"JetBrains Mono"', 'monospace']
            }
          }
        }
      };
    } catch(e) {}
  </script>
  <style>
    *, *::before, *::after { box-sizing: border-box; margin: 0; padding: 0; }
    html, body { width: 100%; min-height: 100vh; }
    body { background: #07070a; color: #f8fafc; font-family: 'Plus Jakarta Sans', system-ui, sans-serif; -webkit-font-smoothing: antialiased; }
  </style>
</head>
<body>
  <div id="sandbox-head-styles"></div>
  <div id="sandbox-app"></div>

  <script>
    // Re-execute <script> tags that innerHTML silently ignores
    function runScripts(container) {
      try {
        container.querySelectorAll('script').forEach(function(old) {
          try {
            var fresh = document.createElement('script');
            Array.from(old.attributes).forEach(function(a) {
              try { fresh.setAttribute(a.name, a.value); } catch(e) {}
            });
            fresh.textContent = old.textContent;
            old.parentNode.replaceChild(fresh, old);
          } catch(e) {}
        });
      } catch(e) {}
    }

    // Inject <style> and <link> tags from the AI-generated <head>
    function injectHeadAssets(html) {
      try {
        var headMatch = html.match(/<head[^>]*>([\s\S]*?)<\/head>/i);
        if (!headMatch) return;
        var headHtml = headMatch[1];
        var container = document.getElementById('sandbox-head-styles');
        if (!container) return;
        container.innerHTML = '';

        // <style> blocks — create programmatically, never via innerHTML
        var styles = headHtml.match(/<style[^>]*>([\s\S]*?)<\/style>/gi) || [];
        styles.forEach(function(s) {
          try {
            var css = s.replace(/<\/?style[^>]*>/gi, '');
            var el = document.createElement('style');
            el.textContent = css;
            container.appendChild(el);
          } catch(e) {}
        });

        // <link> tags — create programmatically to avoid innerHTML issues
        var linkRe = /<link([^>]*)>/gi;
        var linkMatch;
        while ((linkMatch = linkRe.exec(headHtml)) !== null) {
          try {
            var attrStr = linkMatch[1];
            if (!/fonts\.googleapis\.com|fonts\.gstatic\.com|stylesheet/i.test(attrStr)) continue;
            var el = document.createElement('link');
            var attrRe = /(\w[\w-]*)=["']([^"']*)["']/g;
            var am;
            while ((am = attrRe.exec(attrStr)) !== null) {
              try { el.setAttribute(am[1], am[2]); } catch(e) {}
            }
            container.appendChild(el);
          } catch(e) {}
        }
      } catch(e) {}
    }

    // Extract <body> content from a full HTML document string
    function extractBody(html) {
      try {
        var m = html.match(/<body[^>]*>([\s\S]*?)<\/body>/i);
        if (m) return m[1];
        var headEnd = html.search(/<\/head>/i);
        if (headEnd !== -1) return html.slice(headEnd + 7);
      } catch(e) {}
      return html;
    }

    window.addEventListener('message', function(event) {
      try {
        var data = event.data;
        if (!data || !data.htmlCode) return;
        injectHeadAssets(data.htmlCode);
        var app = document.getElementById('sandbox-app');
        if (!app) return;
        app.innerHTML = extractBody(data.htmlCode);
        runScripts(app);
        try { lucide.createIcons(); } catch(e) {}
      } catch(e) {}
    });

    // Notify parent window (whether popup chat frame or separate tab page) that sandbox is ready!
    try {
      window.parent.postMessage({ sandboxReady: true }, '*');
    } catch(e) {
      console.error('[Sandbox] Failed sending sandboxReady handshake:', e);
    }
  </script>
</body>
</html>
Phase 2: Resolving the Manifest V3 Sandbox CSP Config Error
If you encountered the warning Invalid value for 'content_security_policy.sandbox', this is because under Chrome Extension Manifest V3, sandbox CSP parameters have changed.
You must define "sandbox" at the top-level mapping under keys, and style "content_security_policy" correctly with the mandatory directive structure. Tell Replit to replace the sandbox section of your manifest.json with this configuration block:
code
JSON
{
  "manifest_version": 3,
  "name": "SnapToAI Extension",
  "version": "3.5",
  "sandbox": {
    "pages": ["sandbox.html"]
  },
  "content_security_policy": {
    "extension_pages": "script-src 'self'; object-src 'self'",
    "sandbox": "sandbox allow-scripts allow-forms allow-popups allow-modals; script-src 'self' 'unsafe-inline' 'unsafe-eval'; child-src 'self';"
  }
}
Note: In MV3, the CSP value for "sandbox" MUST begin with the sandbox keyword followed by the allowed flags (like allow-scripts, etc.). If you omit this first clause, Chrome's manifest parser will reject the bundle.
