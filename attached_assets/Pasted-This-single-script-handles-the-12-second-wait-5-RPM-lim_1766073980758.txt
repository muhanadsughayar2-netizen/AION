This single script handles the 12-second wait (5 RPM limit), shrinks your images (token limit), and auto-detects code vs. images to give the best answer. It is the "All-in-One" solution for your Chrome extension.

JavaScript

/**
 * Snap To AI - Master Engine (Free Tier Optimized)
 * All-in-one: Image Resizing + Auto-Detection + Request Queueing
 */

const GEMINI_API_KEY = "YOUR_API_KEY_HERE"; 
const MODEL_NAME = "gemini-2.0-flash"; // The best free vision model in 2025

// --- 1. THE FREE TIER GATEKEEPER ---
// Free tier allows 5 requests per minute. We force a 12.5s gap to stay 100% safe.
class AIQueue {
    constructor() { this.queue = []; this.isWaiting = false; }
    async add(task) {
        return new Promise((resolve, reject) => {
            this.queue.push({ task, resolve, reject });
            this.process();
        });
    }
    async process() {
        if (this.isWaiting || this.queue.length === 0) return;
        this.isWaiting = true;
        const { task, resolve, reject } = this.queue.shift();
        try { resolve(await task()); } catch (e) { reject(e); }
        // Force the 12.5 second cooldown before next free request
        setTimeout(() => { this.isWaiting = false; this.process(); }, 12500);
    }
}
const aiQueue = new AIQueue();

// --- 2. THE IMAGE OPTIMIZER ---
// Shrinks the snip to 768px max to fit in the free "Low Res" (280 token) bucket.
async function prepareImage(base64) {
    return new Promise((resolve) => {
        const img = new Image();
        img.src = "data:image/png;base64," + base64;
        img.onload = () => {
            const canvas = document.createElement('canvas');
            const scale = Math.min(768 / img.width, 768 / img.height, 1);
            canvas.width = img.width * scale;
            canvas.height = img.height * scale;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
            // JPEG at 70% quality is the sweet spot for AI readability & speed
            resolve(canvas.toDataURL('image/jpeg', 0.7).split(',')[1]);
        };
    });
}

// --- 3. THE MAGIC ANALYZER ---
// This is the function you call when the user clicks the Star icon.
async function analyzeSnip(rawBase64) {
    // 1. Shrink and optimize image locally (0 tokens used)
    const optimizedBase64 = await prepareImage(rawBase64);

    // 2. Queue the request to avoid "Quota Exceeded" errors
    return aiQueue.add(async () => {
        const url = `https://generativelanguage.googleapis.com/v1beta/models/${MODEL_NAME}:generateContent?key=${GEMINI_API_KEY}`;
        
        const payload = {
            system_instruction: { 
                parts: [{ text: "You are a visual assistant. If you see code, explain the logic and find bugs. If you see a UI or image, provide a summary and 3 bullet points. Be extremely concise." }]
            },
            contents: [{ 
                parts: [
                    { text: "Analyze this snip." }, 
                    { inline_data: { mime_type: "image/jpeg", data: optimizedBase64 } }
                ] 
            }],
            generationConfig: {
                media_resolution: "LOW", // Mandatory for staying on Free Tier
                maxOutputTokens: 500
            }
        };

        const response = await fetch(url, { method: 'POST', body: JSON.stringify(payload) });
        const result = await response.json();
        
        if (result.error) throw new Error(result.error.message);
        return result.candidates[0].content.parts[0].text;
    });
}

Why this is your "Final" Solution:
Guaranteed Free: By forcing the 12.5s delay, you never send requests too fast for the free tier.

Zero Token Waste: Resizing and setting media_resolution: LOW keeps every image at exactly 280 tokens, meaning you can do hundreds of snips a day without hitting the limit.

Auto-Context: The system_instruction tells the AI to look for code automatically, so you don't need different buttons for "Text" vs "Code."
