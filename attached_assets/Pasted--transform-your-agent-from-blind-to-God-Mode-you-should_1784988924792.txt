async function readDocContent(tabId) {
  // 1. Attach debugger if not already attached
  await chrome.debugger.attach({ tabId }, "1.3");

  // 2. Capture the Full Accessibility Tree (The Rosetta Stone)
  const { nodes } = await chrome.debugger.sendCommand({ tabId }, "Accessibility.getFullAXTree");

  // 3. Filter for text-bearing nodes (Headings, Paragraphs, Cells)
  const content = nodes
    .filter(node => ["paragraph", "heading", "cell", "text"].includes(node.role?.value))
    .map(node => node.name?.value || "")
    .filter(text => text.trim().length > 0)
    .join("\n");

  await chrome.debugger.detach({ tabId });
  return content || "No readable content found. Ensure the tab is active.";
}

async function goToCell(tabId, cellReference) {

  const nameBoxSelector = "#t-name-box, input[aria-label='Name Box']";
  
  // 1. Click the Name Box to focus

  // 2. Type the cell (e.g., 'B3') and press Enter
  for (const char of cellReference) {
    await chrome.debugger.sendCommand({ tabId }, "Input.dispatchKeyEvent", {
      type: "keyDown", text: char, unmodifiedText: char
    });
  }
  await chrome.debugger.sendCommand({ tabId }, "Input.dispatchKeyEvent", {
    type: "keyDown", windowsVirtualKeyCode: 13, nativeVirtualKeyCode: 13 // Enter key
  });

  return `Focused on cell ${cellReference}. You can now read the value from the Formula Bar.`;
}
