chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === 'get_page_text') {
    // We grab the title and the text content
    const pageData = {
      title: document.title,
      text: document.body.innerText.slice(0, 15000) // Get first 15k chars to stay safe
    };
    sendResponse(pageData);
  }
  return true; 
});

async function handleUpload(preferredPlatform = 'auto', selectedSnaps = null) {
  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });

  // 1. Ask content.js for the text
  let pageData = null;
  try {
    pageData = await chrome.tabs.sendMessage(tab.id, { action: 'get_page_text' });
  } catch (err) {
    console.log("Could not get text, using images only.");
  }

  // 2. Decide: Is there enough text to be useful?
  const isHybrid = pageData && pageData.text.length > 800;

  // 3. Save to the 'Session Locker'
  await chrome.storage.session.set({
    aiText: isHybrid ? pageData.text : null,
    aiTitle: isHybrid ? pageData.title : null,
    payloadMode: isHybrid ? 'hybrid' : 'images'
  });

  // 4. Continue with your existing code to open Gemini/ChatGPT...
  // (Don't change the part that opens the tab!)
}
4. Update the AI Site Logic (The Hands)
This is the part that actually pastes the text into Gemini.

Where: Add this to the section of content.js (or a separate file if you have one) that runs on gemini.google.com.

JavaScript

async function autoPasteText() {
  const data = await chrome.storage.session.get(['aiText', 'aiTitle', 'payloadMode']);
  
  if (data.payloadMode === 'hybrid' && data.aiText) {
    const input = document.querySelector('div[role="textbox"], textarea');
    if (input) {
      const prompt = `SOURCE: ${data.aiTitle}\n\nTEXT CONTENT:\n${data.aiText}\n\n--- INSTRUCTION: Analyze the images using the text above for context. ---`;
      
      input.innerText = prompt;
      input.dispatchEvent(new Event('input', { bubbles: true }));
      
      // Clear the locker so it doesn't paste again on refresh
      chrome.storage.session.remove(['aiText', 'aiTitle', 'payloadMode']);
    }
  }
}

// Check for the input box every second until it appears
const pasteInterval = setInterval(() => {
  if (document.querySelector('div[role="textbox"], textarea')) {
    autoPasteText();
    clearInterval(pasteInterval);
  }
}, 1000);
