Step 1: The Replit Backend (index.js)
What this does: It uses the "Flash" model (the cheapest) and forces the AI to be brief (saving tokens/money). It combines the "color scan" and "analysis" into one single request.

// POST /chat-verdict
app.post('/chat-verdict', async (req, res) => {
  const { imageBase64, chatQuery, selectedItem } = req.body;

  try {
    // 1. Use gemini-1.5-flash (Fastest & Lowest Cost)
    const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });

    const systemPrompt = `
      You are the "SnapToAI Verdict Engine". 
      Context: User is looking at ${selectedItem || 'this image'} and asked: "${chatQuery}".
      
      TASK: 
      1. Analyze image for prices/quality. Compare specific items (e.g. tablecloth vs placemats).
      2. Give a definitive "YES/BUY", "NO/AVOID", or "HOLD/WAIT" verdict.
      3. No generic advice. Be specific to the items seen.
      
      OUTPUT ONLY VALID JSON:
      {
        "verdict": "BUY",
        "header": "⚖️ THE VERDICT: [INSERT]",
        "insiderSecret": "One specific detail found in the image.",
        "priceCrush": "Short comparison vs market.",
        "confidenceScore": "90%",
        "glowColor": "gold", 
        "action": "ADD TO CART"
      }
      (glowColor: 'gold' for buy, 'red' for avoid, 'green' for neutral).
    `;

    const parts = [
      { text: systemPrompt },
      { inlineData: { data: imageBase64, mimeType: "image/png" } }
    ];

    // 2. Strict token limit (300) to keep costs minimal
    const result = await model.generateContent({
      contents: [{ role: "user", parts }],
      generationConfig: { maxOutputTokens: 300, temperature: 0.7 }
    });

    const text = result.response.text();
    const cleanJson = text.replace(/```json|```/g, '').trim();
    res.json(JSON.parse(cleanJson));

  } catch (err) {
    console.error("Verdict Error:", err);
    res.status(500).json({ error: "Verdict failed." });
  }
});
