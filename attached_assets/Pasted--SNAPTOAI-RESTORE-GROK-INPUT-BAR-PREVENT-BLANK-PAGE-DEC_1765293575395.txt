// ==== SNAPTOAI — RESTORE GROK INPUT BAR & PREVENT BLANK PAGE (DEC 2025 FIX) ====
function restoreGrokInput() {
  if (!window.location.hostname.includes('grok.x.ai')) return;

  console.log('SnapToAI: Restoring Grok input bar — no more blank voids!');

  // Safety check: If chat area is empty/missing, force reload the input
  const chatContainer = document.querySelector('main, [role="main"], div[data-testid="conversation"]') ||
                        document.querySelector('div[class*="chat-container"], div[class*="conversation-root"]');
  
  if (!chatContainer) {
    console.log('SnapToAI: Chat container missing — triggering Grok reload');
    // Simulate a soft refresh to bring back the UI without full page reload
    const event = new Event('visibilitychange', { bubbles: true });
    document.dispatchEvent(event);
    setTimeout(() => window.location.reload(), 500); // Nuclear option only if needed
    return;
  }

  // Smart hide: Only nuke duplicate/fake inputs, NEVER the real one
  const allInputs = document.querySelectorAll('textarea, div[contenteditable="true"]');
  const realInput = allInputs.find(el => 
    el.placeholder?.toLowerCase().includes('message grok') || 
    el.getAttribute('data-testid') === 'message-input' ||
    el.closest('[role="textbox"]') // Grok's real chat box wrapper
  );

  if (realInput) {
    // Ensure it's visible and functional
    realInput.style.display = 'block';
    realInput.style.visibility = 'visible';
    realInput.style.opacity = '1';
    realInput.focus(); // Bonus: Auto-focus so you're ready to roll

    // Hide ONLY ghosts/duplicates (not the real deal)
    allInputs.forEach(input => {
      if (input !== realInput && (input.style.display === 'none' || input.id.includes('snap-clone'))) {
        input.remove(); // Bye, fakes
      }
    });

    console.log('SnapToAI: Real input restored — chat bar back online!');
  } else {
    // Fallback: Poll until Grok loads it (they lazy-load sometimes)
    const pollInterval = setInterval(() => {
      const freshInput = document.querySelector('textarea[placeholder*="Message"], [data-testid="message-input"]');
      if (freshInput) {
        clearInterval(pollInterval);
        restoreGrokInput(); // Recursive magic
      }
    }, 500);

    setTimeout(() => clearInterval(pollInterval), 10000); // Stop after 10s
  }

  // Auto-scroll to bottom on load (so you see your existing chat, not blank top)
  setTimeout(() => {
    chatContainer.scrollTop = chatContainer.scrollHeight;
  }, 800);
}

// Hook it into page load + mutations (so it survives Grok's dynamic BS)
if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', restoreGrokInput);
} else {
  restoreGrokInput();
}

// MutationObserver to watch for input zaps and auto-restore
const observer = new MutationObserver((mutations) => {
  let inputVanished = false;
  mutations.forEach((mutation) => {
    if (mutation.removedNodes.length > 0) {
      mutation.removedNodes.forEach(node => {
        if (node.nodeType === 1 && (node.matches('textarea, [contenteditable="true"]') || node.querySelector('textarea'))) {
          inputVanished = true;
        }
      });
    }
  });
  if (inputVanished) {
    setTimeout(restoreGrokInput, 100);
  }
});
observer.observe(document.body, { childList: true, subtree: true });
