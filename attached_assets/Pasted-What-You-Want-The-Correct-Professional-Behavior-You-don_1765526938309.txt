What You Want (The Correct, Professional Behavior)
You don't want the current "Resize Canvas" that scales the entire image (making it smaller/bigger and blurry).
Instead, you want a visual crop tool like GoFullPage, Snagit, or Markup Hero:

User clicks a "Crop" button.
A draggable/resizable blue rectangle appears on the image (with 8 handles: corners + sides).
User drags the edges/handles to select the area they want to keep.
Everything outside the rectangle is dimmed (dark overlay).
When user clicks "Done Cropping" or double-clicks → the image is permanently cropped to that rectangle.
Everything outside disappears — clean, professional result.

This is exactly how premium tools work — and it's much better than the current resize.

JavaScriptfunction handleMouseDown(e) {
  if (!isCropMode) {
    // Your existing annotation logic here
    return;
  }
  
  const pos = getMousePos(e);
  const { x, y, width, height } = cropRect;
  const handleSize = 20;
  
  // Check if clicking a handle
  cropHandle = null;
  
  if (Math.abs(pos.x - x) < handleSize && Math.abs(pos.y - y) < handleSize) cropHandle = 'tl';
  else if (Math.abs(pos.x - (x + width)) < handleSize && Math.abs(pos.y - y) < handleSize) cropHandle = 'tr';
  else if (Math.abs(pos.x - x) < handleSize && Math.abs(pos.y - (y + height)) < handleSize) cropHandle = 'bl';
  else if (Math.abs(pos.x - (x + width)) < handleSize && Math.abs(pos.y - (y + height)) < handleSize) cropHandle = 'br';
  else if (Math.abs(pos.y - y) < handleSize && pos.x > x && pos.x < x + width) cropHandle = 't';
  else if (Math.abs(pos.x - (x + width)) < handleSize && pos.y > y && pos.y < y + height) cropHandle = 'r';
  else if (Math.abs(pos.y - (y + height)) < handleSize && pos.x > x && pos.x < x + width) cropHandle = 'b';
  else if (Math.abs(pos.x - x) < handleSize && pos.y > y && pos.y < y + height) cropHandle = 'l';
  else if (pos.x > x && pos.x < x + width && pos.y > y && pos.y < y + height) cropHandle = 'move';
  
  cropStartX = pos.x;
  cropStartY = pos.y;
  
  if (cropHandle) {
    canvas.style.cursor = getCropCursor(cropHandle);
  }
}

function handleMouseMove(e) {
  if (!isCropMode) return;
  
  const pos = getMousePos(e);
  
  if (cropHandle) {
    const dx = pos.x - cropStartX;
    const dy = pos.y - cropStartY;
    
    if (cropHandle.includes('l') || cropHandle === 'move') cropRect.x += dx;
    if (cropHandle.includes('t') || cropHandle === 'move') cropRect.y += dy;
    if (cropHandle.includes('r')) cropRect.width += dx;
    if (cropHandle.includes('b')) cropRect.height += dy;
    if (cropHandle.includes('l')) cropRect.width -= dx;
    if (cropHandle.includes('t')) cropRect.height -= dy;
    
    // Prevent negative size
    if (cropRect.width < 50) cropRect.width = 50;
    if (cropRect.height < 50) cropRect.height = 50;
    if (cropRect.x < 0) cropRect.x = 0;
    if (cropRect.y < 0) cropRect.y = 0;
    if (cropRect.x + cropRect.width > canvas.width) cropRect.width = canvas.width - cropRect.x;
    if (cropRect.y + cropRect.height > canvas.height) cropRect.height = canvas.height - cropRect.y;
    
    cropStartX = pos.x;
    cropStartY = pos.y;
  } else {
    // Change cursor when hovering handles
    const { x, y, width, height } = cropRect;
    const handleSize = 20;
    let newCursor = 'default';
    if (Math.abs(pos.x - x) < handleSize && Math.abs(pos.y - y) < handleSize) newCursor = 'nwse-resize';
    else if (Math.abs(pos.x - (x + width)) < handleSize && Math.abs(pos.y - y) < handleSize) newCursor = 'nesw-resize';
    // ... add others
    else if (pos.x > x && pos.x < x + width && pos.y > y && pos.y < y + height) newCursor = 'move';
    canvas.style.cursor = newCursor;
  }
  
  redraw();
}

function getCropCursor(handle) {
  if (handle === 'move') return 'move';
  if (handle === 'tl' || handle === 'br') return 'nwse-resize';
  if (handle === 'tr' || handle === 'bl') return 'nesw-resize';
  if (handle === 't' || handle === 'b') return 'ns-resize';
  if (handle === 'l' || handle === 'r') return 'ew-resize';
  return 'default';
}

}
