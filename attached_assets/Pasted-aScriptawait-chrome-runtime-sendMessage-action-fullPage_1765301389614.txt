→ you do this 30–100 times in a row with only 50–100ms delays
→ Chrome throttles and fails 90% of them → broken full-page screenshot
The Fix (100% Working – Used by GoFullPage, FireShot, etc.)
Replace your current capture loop with this exact, battle-tested version that respects Chrome’s rate limit:
JavaScript// === REPLACE THE ENTIRE CAPTURE LOOP WITH THIS (inside performFullPageCapture) ===
let lastScrollTop = -1;
let captureCount = 0;
const maxCaptures = 100;
let consecutiveFails = 0;
const MAX_CONSECUTIVE_FAILS = 5;

while (captureCount < maxCaptures && consecutiveFails < MAX_CONSECUTIVE_FAILS) {
  const currentScrollTop = getScrollTop();

  // Update progress
  const maxScroll = getMaxScroll();
  const progress = maxScroll > 0 ? Math.round((currentScrollTop / maxScroll) * 100) : 50;
  updateOverlayProgress(Math.min(99, progress), captureCount + 1, Math.ceil(maxScroll / stepHeight) + 2);

  // HIDE overlay
  overlay.style.visibility = 'hidden';
  await new Promise(r => setTimeout(r, 80)); // let render settle

  // === CRITICAL: ADD 600–800ms DELAY BETWEEN CAPTURES ===
  const response = await new Promise(resolve => {
    const timeout = setTimeout(() => {
      chrome.runtime.sendMessage(
        { action: 'fullPageCaptureStep', tabId: tabId },
        resp => {
          clearTimeout(timeout);
          resolve(resp || { success: false, error: 'timeout' });
        }
      );
    }, 100);

    // Force minimum delay to respect Chrome quota
    setTimeout(() => {}, 700); // ← THIS IS THE KEY
  });

  overlay.style.visibility = 'visible';

  if (response?.success && response.dataUrl) {
    screenshots.push({ dataUrl: response.dataUrl, scrollY: currentScrollTop, index: captureCount });
    console.log(`[SnapToAI] Captured ${captureCount + 1}`);
    consecutiveFails = 0; // reset on success
  } else {
    console.warn(`[SnapToAI] Capture ${captureCount + 1} failed:`, response?.error || 'no response');
    consecutiveFails++;
    // Still count it so we don't loop forever
  }

  captureCount++;

  // Stop if at bottom
  if (currentScrollTop === lastScrollTop && captureCount > 2) break;
  if (currentScrollTop >= getMaxScroll() - 20) break;

  lastScrollTop = currentScrollTop;
  safeScrollBy(stepHeight);

  // Wait for content to load + Chrome rate limit
  await waitForScrollStabilization(scrollContainer, useContainerScroll);
  // CRITICAL: Wait at least 700ms before next capture
  await new Promise(r => setTimeout(r, 700)); // ← THIS LINE SAVES YOU
}
