PART 1: CSS Injection Function (Lines 164-214 in content.js)
// Force Google Docs to render full content by injecting expansion styles
function forceGoogleDocsFullRender() {
  if (!location.hostname.includes('docs.google.com')) return;  // ← Only runs on Google Docs
  
  // Remove any existing style to avoid duplicates
  const existing = document.getElementById('snaptoai-docs-styles');
  if (existing) existing.remove();
  
  const style = document.createElement('style');
  style.id = 'snaptoai-docs-styles';
  style.textContent = `
    /* Force full expansion for Google Docs - like GoFullPage */
    body, html {
      height: auto !important;           // ← Let height expand infinitely
      min-height: auto !important;
      max-height: none !important;
      overflow: visible !important;      // ← Show all content (don't hide in scrollbars)
      position: static !important;
      background-color: #fff !important;
    }
    .kix-appview-editor, .docs-editor, .kix-page-canvas, .kix-canvas-tile-content, canvas {
      height: auto !important;
      min-height: auto !important;
      max-height: none !important;
      overflow: visible !important;      // ← Google Docs containers show full content
      position: static !important;
      display: block !important;
      visibility: visible !important;
    }
    /* Hide sticky/fixed elements to avoid duplication */
    [style*="position: sticky"], [style*="position: fixed"] {
      display: none !important;          // ← Remove headers/toolbars that repeat
      visibility: hidden !important;
    }
    /* Keep header/nav but fix docs-specific sticky elements */
    .docs-titlebar, .docs-menubar {
      position: relative !important;     // ← These scroll with content, not fixed
    }
    /* Expand all containers */
    .kix-appview-editor-container, .kix-zoomdocumentplugin-outer {
      height: auto !important;
      overflow: visible !important;
      position: relative !important;
    }
  `;
  document.head.appendChild(style);  // ← Inject the CSS into page
  
  // Force layout recalculation
  document.body.offsetHeight;  // ← Triggers browser to recalculate layout
  
  console.log('[SnapToAI] Google Docs styles injected for full render');
}

What this does:

Creates a <style> tag with CSS rules
height: auto → lets the page grow to full document height instead of staying fixed
overflow: visible → shows all content instead of hiding it behind scrollbars
position: static/relative → makes headers scroll with content instead of staying fixed
Injects into <head> so it applies to the entire page
document.body.offsetHeight forces browser to recalculate layout

async function safeFullPageCapture(tabId) {
  try {
    // *** STEP 1: INJECT CSS ***
    forceGoogleDocsFullRender();  // ← Expand Google Docs
    
    // *** STEP 2: WAIT FOR CSS TO APPLY ***
    if (location.hostname.includes('docs.google.com')) {
      await new Promise(r => setTimeout(r, 300));  // ← Wait 300 milliseconds
    }
    
    // *** STEP 3: DO NORMAL SCROLL-CAPTURE ***
    const result = await performFullPageCapture(tabId);  // ← This does the scrolling & screenshots
    
    // *** STEP 4: CLEANUP ***
    cleanupGoogleDocsStyles();  // ← Remove the CSS injection
    
    return result;
  } catch (error) {
    // ... error handling
    cleanupGoogleDocsStyles();  // ← Also cleanup if error
  }
}

What this does:

Calls forceGoogleDocsFullRender() → injects CSS
Waits 300ms for browser to render the expanded layout
Calls performFullPageCapture() → the EXISTING scrolling code (explained below)
Cleans up the CSS
