/**
 * 1. Generates Flashcards using your existing Gemini connection
 * @param {string} textContext - The text from the screenshot/page to learn from
 */
async function generateFlashcards(textContext) {
  const container = document.getElementById('chat-output'); // Or wherever your chat is
  
  // Show a loading state
  const loadingId = 'loading-' + Date.now();
  container.innerHTML += `<div id="${loadingId}" class="message ai">Generating study cards...</div>`;

  try {
    // THE PROMPT: We strictly ask for JSON format
    const prompt = `
      Based on the following text, generate 5 study flashcards (Questions and Answers).
      Strictly return ONLY a valid JSON array in this format:
      [{"front": "Question", "back": "Answer"}]
      Do not add markdown formatting or conversational text.
      
      TEXT TO STUDY:
      ${textContext.substring(0, 3000)} // Limit context to save tokens
    `;

    // Call your existing API function (assuming it's named callGeminiAPI)
    const rawResponse = await callGeminiAPI(prompt);
    
    // Clean up the response (remove ```json blocks if Gemini adds them)
    const cleanJson = rawResponse.replace(/```json|```/g, '').trim();
    const cards = JSON.parse(cleanJson);

    // Remove loading message
    document.getElementById(loadingId).remove();
    
    // Render the cards
    renderFlashcards(cards);

  } catch (error) {
    console.error("Flashcard Error:", error);
    document.getElementById(loadingId).innerText = "Error generating cards. Please try again.";
  }
}

/**
 * 2. Renders the 3D Cards into the Chat UI
 */
function renderFlashcards(cards) {
  const chatBox = document.getElementById('chat-output');
  const uniqueId = 'cards-' + Date.now();
  
  // Create a wrapper for this set of cards
  const wrapper = document.createElement('div');
  wrapper.className = 'flashcard-wrapper';
  wrapper.id = uniqueId;

  // Add the Save PDF Button
  const saveBtn = document.createElement('button');
  saveBtn.className = 'save-pdf-btn';
  saveBtn.innerText = '📄 Download PDF';
  saveBtn.onclick = () => printFlashcards(uniqueId);
  wrapper.appendChild(saveBtn);

  // Create the Grid
  const grid = document.createElement('div');
  grid.className = 'flashcard-grid';

  cards.forEach(card => {
    // HTML Structure for 3D Flip
    const cardHTML = `
      <div class="flashcard-container" onclick="this.classList.toggle('flipped')">
        <div class="flashcard-inner">
          <div class="flashcard-front">
            <div>${card.front}</div>
          </div>
          <div class="flashcard-back">
            <div>${card.back}</div>
          </div>
        </div>
      </div>
    `;
    grid.innerHTML += cardHTML;
  });

  wrapper.appendChild(grid);
  chatBox.appendChild(wrapper);
  
  // Scroll to bottom
  chatBox.scrollTop = chatBox.scrollHeight;
}

/**
 * 3. Prints the cards nicely for cutting
 */
function printFlashcards(wrapperId) {
  const content = document.getElementById(wrapperId).querySelector('.flashcard-grid').outerHTML;
  
  const printWindow = window.open('', '', 'height=600,width=800');
  printWindow.document.write('<html><head><title>Study Flashcards</title>');
  printWindow.document.write(`
    <style>
      body { font-family: sans-serif; padding: 20px; }
      .flashcard-grid { display: block; }
      .flashcard-container { 
        border: 2px dashed #333; 
        margin: 10px; 
        padding: 20px; 
        page-break-inside: avoid;
        display: inline-block;
        width: 45%;
        height: 150px;
        vertical-align: top;
      }
      .flashcard-front { font-weight: bold; border-bottom: 1px solid #ccc; margin-bottom: 5px; padding-bottom: 5px; display: block;}
      .flashcard-back { display: block; color: #555; }
      .flashcard-inner, .flashcard-front, .flashcard-back { position: static; transform: none !important; display: block; height: auto;}
    </style>
  `);
  printWindow.document.write('</head><body>');
  printWindow.document.write('<h2>Flashcards (Cut along dashed lines)</h2>');
  printWindow.document.write(content);
  printWindow.document.write('</body></html>');
  
  printWindow.document.close();
  printWindow.print();
}
