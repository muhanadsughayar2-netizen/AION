 * Atomic Saver: Guarantees snaps and metadata are saved together.

async function handleSaveSnap(request) {
  try {
    const store = await chrome.storage.local.get(['snaps', 'snapMetadata']);
    let snaps = store.snaps || [];
    let metadata = store.snapMetadata || [];

    // SELF-HEALING: FIX array length differences before adding new data
    if (metadata.length !== snaps.length) {
      if (metadata.length < snaps.length) {
        const diff = snaps.length - metadata.length;
        metadata = [...metadata, ...Array(diff).fill(null)];
      } else {
        metadata = metadata.slice(0, snaps.length);
      }
    }

    // Prepare the new Metadata Object (Guarantees {isSnip: true} is saved)
    let newMeta = null;
    if (request.chunkInfo) {
      newMeta = {
        isChunk: true,
        part: request.chunkInfo.part,
        totalParts: request.chunkInfo.totalParts,
        captureGroupId: request.chunkInfo.captureGroupId,
        smartName: request.smartName || 'Full Page'
      };
    } else if (request.isSnip) {
      newMeta = { isSnip: true, timestamp: Date.now() };
    } 

    // ATOMIC UPDATE: Save BOTH in one call
    snaps.push(request.dataUrl);
    metadata.push(newMeta);

    await chrome.storage.local.set({
      snaps: snaps,
      snapMetadata: metadata
    });

    return { success: true, count: snaps.length };

  } catch (error) {
    return { success: false, error: error.message };
  }
}

const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

// Helper 1: Pings the background script to take a screenshot and returns the data URL.
async function captureVisibleTab() {
  const request = { action: 'captureVisible' };

  return new Promise((resolve, reject) => {
    chrome.runtime.sendMessage(request, (response) => {
      if (chrome.runtime.lastError) {
        return reject(chrome.runtime.lastError);
      }
      if (response && response.dataUrl) {
        resolve(response.dataUrl);
      } else {
        reject(new Error('Background script failed to return a dataUrl.'));
      }
    });
  });
}

// Helper 2: Sends data to background and WAITS for save confirmation (THROTTLING)
async function sendSnapToBackground(payload) {
  return new Promise((resolve, reject) => {
    chrome.runtime.sendMessage(payload, (response) => {
      if (chrome.runtime.lastError) {
        return reject(chrome.runtime.lastError);
      }
      if (response && response.success) {
        resolve(response);
      } else {
        reject(new Error(response?.error || 'Unknown save error'));
      }
    });
  });
}

// --- 3.2: STICKY ELEMENT FIX LOGIC ---

let fixedElementsState = [];

function findAndHideFixedElements() {
    fixedElementsState = [];
    document.querySelectorAll('*').forEach(el => {
        const style = window.getComputedStyle(el);
        if (style.position === 'fixed' || style.position === 'sticky') {
            // Only hide visible, non-tiny fixed elements
            if (el.offsetWidth > 0 && el.offsetHeight > 0 && el.offsetHeight < window.innerHeight * 0.9) {
                
                fixedElementsState.push({
                    element: el,
                    originalVisibility: el.style.visibility,
                    originalPosition: el.style.position,
                    originalTop: el.style.top,
                });
                
                el.style.visibility = 'hidden';
                el.style.top = '-9999px'; 
                el.style.setProperty('position', 'absolute', 'important'); // Ensure it doesn't interfere
            }
        }
    });
}

function restoreFixedElements() {
    fixedElementsState.forEach(state => {
        state.element.style.visibility = state.originalVisibility;
        state.element.style.position = state.originalPosition; // Restores 'fixed' or 'sticky'
        state.element.style.top = state.originalTop;
    });
    fixedElementsState = [];
}
