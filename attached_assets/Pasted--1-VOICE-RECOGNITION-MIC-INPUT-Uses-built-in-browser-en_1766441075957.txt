/**
 * 1. VOICE RECOGNITION (MIC INPUT)
 * Uses built-in browser engine (Free)
 */
const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
let recognition;

if (SpeechRecognition) {
  recognition = new SpeechRecognition();
  recognition.continuous = false;
  recognition.lang = 'en-US';

  recognition.onresult = (event) => {
    const transcript = event.results[0][0].transcript;
    generateFlashcards(transcript);
  };

  recognition.onstart = () => {
    const star = document.getElementById('star-mic');
    if(star) star.classList.add('listening-pulse');
  };

  recognition.onend = () => {
    const star = document.getElementById('star-mic');
    if(star) star.classList.remove('listening-pulse');
  };
}

function toggleVoiceInput() {
  if (!recognition) return;
  try { recognition.start(); } catch (e) { recognition.stop(); }
}

/**
 * 2. SPEECH SYNTHESIS (READ ALOUD)
 */
function speakText(text) {
  if ('speechSynthesis' in window) {
    const utterance = new SpeechSynthesisUtterance(text);
    utterance.rate = 1.0;
    utterance.pitch = 1.1;
    window.speechSynthesis.speak(utterance);
  }
}
