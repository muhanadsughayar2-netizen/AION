    function updateConsoleStats() {
      const count = activePayloads.size;
      const countLabel = document.getElementById('active-count');
      const latencyLabel = document.getElementById('latency-val');
      const loadLabel = document.getElementById('load-val');
      const gauge = document.getElementById('console-gauge');
      const metric = document.getElementById('console-metric');
      const runBtn = document.getElementById('run-inference-btn');

      if (countLabel) countLabel.innerText = `${count} / 8`;

      if (count === 0) {
        if (latencyLabel) latencyLabel.innerText = `-- ms`;
        if (loadLabel) loadLabel.innerText = `0.0 GB`;
        if (metric) metric.innerText = `0%`;
        if (gauge) gauge.style.strokeDashoffset = "251.2";
        if (runBtn) {
          runBtn.disabled = true;
          runBtn.className = "w-full py-3 rounded-lg bg-white/5 border border-white/10 text-gray-500 font-semibold text-sm transition-all flex items-center justify-center gap-2 cursor-not-allowed";
        }
        return;
      }

      // Calculate totals
      let totalLatency = 0;
      let totalLoad = 0;
      activePayloads.forEach(id => {
        totalLatency += payloadSpecs[id].latency;
        totalLoad += payloadSpecs[id].load;
      });

      // Average latency
      const avgLatency = Math.round(totalLatency / count);
      if (latencyLabel) latencyLabel.innerText = `${avgLatency} ms`;
      if (loadLabel) loadLabel.innerText = `${totalLoad.toFixed(1)} GB`;

      // Gauge math
      const percentage = Math.round((count / 8) * 100);
      if (metric) metric.innerText = `${percentage}%`;
      
      const circumference = 2 * Math.PI * 40; // ~251.2
      const offset = circumference - (percentage / 100) * circumference;
      if (gauge) gauge.style.strokeDashoffset = offset;

      // Enable execute button
      if (runBtn) {
        runBtn.disabled = false;
        runBtn.className = "w-full py-3 rounded-lg bg-gradient-to-r from-purple-500 to-pink-500 text-white font-semibold text-sm hover:opacity-90 transition-all flex items-center justify-center gap-2 cursor-pointer shadow-lg shadow-purple-500/10";
      }
    }

      const gauge = document.getElementById('console-gauge');
      const metric = document.getElementById('console-metric');

        if (loadLabel) loadLabel.innerText = `0.0 GB`;

        if (gauge) gauge.style.strokeDashoffset = "251.2";

        return;
      }

      // Calculate totals
      let totalLatency = 0;
      let totalLoad = 0;
      activePayloads.forEach(id => {
        totalLatency += payloadSpecs[id].latency;
        totalLoad += payloadSpecs[id].load;

      // Average latency

      if (metric) metric.innerText = `${percentage}%`;
      
      const circumference = 2 * Math.PI * 40; // ~251.2
      const offset = circumference - (percentage / 100) * circumference;
      if (gauge) gauge.style.strokeDashoffset = offset;

      // Enable execute button
