  <script>
    (function() {
      const appState = {
        wishlist: [],
        ui: {
          wishlistOpen: false,
          mobileMenuOpen: false
        }
      };

      lucide.createIcons();

      const _io = new IntersectionObserver(entries => {
        entries.forEach(e => {
          if (e.isIntersecting) {
            e.target.classList.add('visible');
            _io.unobserve(e.target);
          }
        });
      }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

      document.querySelectorAll('.reveal').forEach(el => _io.observe(el));

      window.toggleWishlist = function() {
        const panel = document.getElementById('wishlist-panel');
        const overlay = document.getElementById('wishlist-overlay');
        appState.ui.wishlistOpen = !appState.ui.wishlistOpen;
        
        if (appState.ui.wishlistOpen) {
          panel.classList.remove('translate-x-full');
          overlay.classList.remove('hidden');
          document.body.style.overflow = 'hidden';
        } else {
          panel.classList.add('translate-x-full');
          overlay.classList.add('hidden');
          document.body.style.overflow = 'auto';
        }
      };

      window.toggleMobileMenu = function() {
        const menu = document.getElementById('mobile-menu');
        appState.ui.mobileMenuOpen = !appState.ui.mobileMenuOpen;
        menu.classList.toggle('hidden', !appState.ui.mobileMenuOpen);
        menu.classList.toggle('flex', appState.ui.mobileMenuOpen);
        document.body.style.overflow = appState.ui.mobileMenuOpen ? 'hidden' : 'auto';
      };

      window.addToWishlist = function(id, name, price, img) {
        if (appState.wishlist.find(item => item.id === id)) {
          window.showToast("Already in Wishlist");
          return;
        }
        appState.wishlist.push({ id, name, price, img });
        window.updateWishlistUI();
        window.showToast(`${name} Added`);
      };

      window.removeFromWishlist = function(id) {
        appState.wishlist = appState.wishlist.filter(item => item.id !== id);
        window.updateWishlistUI();
      };

      window.updateWishlistUI = function() {
        const container = document.getElementById('wishlist-items');
        const badge = document.getElementById('wishlist-badge');
        const totalEl = document.getElementById('wishlist-total');
        
        if (appState.wishlist.length === 0) {
          container.innerHTML = `
            <div class="h-full flex flex-col items-center justify-center text-center opacity-40">
              <i data-lucide="heart" class="w-12 h-12 mb-4"></i>
              <p class="font-heading text-xl">Your wishlist is empty</p>
              <p class="text-xs uppercase tracking-widest mt-2">Discover your next masterpiece</p>
            </div>
          `;
          badge.classList.add('opacity-0');
          totalEl.textContent = '$0';
        } else {
          badge.textContent = appState.wishlist.length;
          badge.classList.remove('opacity-0');
          
          let total = 0;
          container.innerHTML = appState.wishlist.map(item => {
            total += item.price;
            return `
              <div class="flex gap-4 items-center animate-fadeIn">
                <img src="${item.img}" class="w-20 h-16 object-cover rounded-sm border border-white/10">
                <div class="flex-grow">
                  <h4 class="font-heading text-lg">${item.name}</h4>
                  <p class="text-accent text-xs font-bold">$${item.price.toLocaleString()}</p>
                </div>
                <button onclick="window.removeFromWishlist('${item.id}')" class="text-white/20 hover:text-red-500 transition-colors">
                  <i data-lucide="trash-2" class="w-4 h-4"></i>
                </button>
              </div>
            `;
          }).join('');
          totalEl.textContent = '$' + total.toLocaleString();
        }
        lucide.createIcons();
      };

      window.showToast = function(msg) {
        const t = document.getElementById('toast');
        document.getElementById('toast-msg').textContent = msg;
        t.classList.add('show');
        setTimeout(() => t.classList.remove('show'), 3000);
      };

      window.openBooking = function(carName) {
        window.openLegal('Book Test Drive', `
          <p>You are requesting a private test drive for the <strong>${carName}</strong>.</p>
          <p>Please provide your contact details below or use our concierge form. A representative will contact you within 2 business hours to verify credentials and schedule your appointment.</p>
          <div class="mt-8 pt-8 border-t border-white/10">
            <a href="#contact" onclick="window.closeLegal()" class="accent-gradient px-8 py-3 text-white text-xs font-bold uppercase tracking-widest inline-block">Go to Concierge Form</a>
          </div>
        `);
      };

      window.handleContact = function(e) {
        e.preventDefault();
        window.showToast("Inquiry Sent Successfully");
        e.target.reset();
      };

      window.requestCallback = function() {
        if (appState.wishlist.length === 0) {
          window.showToast("Wishlist is empty");
          return;
        }
        window.showToast("Concierge Request Received");
        setTimeout(() => window.toggleWishlist(), 500);
      };

      window.openLegal = function(title, body) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = body;
        document.getElementById('legalModal').style.display = 'flex';
        document.body.style.overflow = 'hidden';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
        document.body.style.overflow = 'auto';
      };

      document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
          e.preventDefault();
          const target = document.querySelector(this.getAttribute('href'));
          if (target) {
            window.scrollTo({
              top: target.getBoundingClientRect().top + window.pageYOffset - 120,
              behavior: 'smooth'
            });
          }
        });
      });

    })();
  </script>
