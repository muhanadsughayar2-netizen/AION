3. Fix for content.js (Shadow DOM Piercing)
This utility allows your agent to "see" through the hidden layers (Shadow DOM) used by sites like Claude and Gemini to hide their text from bots.

// content.js - Shadow DOM Piercing
function findTextDeep(selector) {
  let elements = [];
  const root = document.body;

  function walk(node) {
    if (node.shadowRoot) {
      walk(node.shadowRoot);
    }
    if (node.querySelectorAll) {
      const found = node.querySelectorAll(selector);
      if (found.length > 0) elements.push(...found);
    }
    node.childNodes.forEach(child => walk(child));
  }

  walk(root);
  return elements;
}
