        if (appState.isCartOpen) {
          drawer.classList.add('open');

          drawer.classList.remove('open');

        appState.isMobileMenuOpen = !appState.isMobileMenuOpen;

      window.addToCart = function(id, name, price) {
        const existing = appState.cart.find(item => item.id === id);
        if (existing) {
          existing.qty++;
        } else {
          appState.cart.push({ id, name, price, qty: 1 });
        }
        renderCart();
        window.showToast(`${name} added to bag`);
      };

      window.removeItem = function(id) {
        appState.cart = appState.cart.filter(item => item.id !== id);
        renderCart();
      };

      window.changeQty = function(id, delta) {
        const item = appState.cart.find(i => i.id === id);
        if (item) {
          item.qty += delta;
          if (item.qty <= 0) window.removeItem(id);
          else renderCart();
        }
      };

      function renderCart() {

        const footer = document.getElementById('cart-footer');

        const totalQty = appState.cart.reduce((s, i) => s + i.qty, 0);
        const totalPrice = appState.cart.reduce((s, i) => s + (i.price * i.qty), 0);

    (function() {
      // 1. App State
      const appState = {
        cart: [],
        isCartOpen: false,
        isMobileMenuOpen: false,
        products: [
          {id: 1, name: 'Lavender Dream', price: 24},
          {id: 2, name: 'Sandalwood Noir', price: 32},
          {id: 3, name: 'Rose Petal', price: 28},
          {id: 4, name: 'Vanilla Sky', price: 26},
          {id: 5, name: 'Midnight Oud', price: 38},
          {id: 6, name: 'Ocean Breeze', price: 29}
        ]
      };

      // Clear stale cart on load
      localStorage.removeItem('cart_bloomandco');

      // 2. UI Functions
      window.toggleCart = function() {
        appState.isCartOpen = !appState.isCartOpen;
        const drawer = document.getElementById('cart-drawer');
        const overlay = document.getElementById('cart-overlay');
        if (appState.isCartOpen) {
          drawer.classList.add('open');
          overlay.classList.add('opacity-100', 'pointer-events-auto');
        } else {
          drawer.classList.remove('open');
          overlay.classList.remove('opacity-100', 'pointer-events-auto');
        }
      };

      window.toggleMobileMenu = function() {
        appState.isMobileMenuOpen = !appState.isMobileMenuOpen;
        const menu = document.getElementById('mobile-menu');
        menu.style.transform = appState.isMobileMenuOpen ? 'translateX(0)' : 'translateX(100%)';
      };

      window.addToCart = function(id, name, price) {
        const existing = appState.cart.find(item => item.id === id);
        if (existing) {
          existing.qty++;
        } else {
          appState.cart.push({ id, name, price, qty: 1 });
        }
        renderCart();
        window.showToast(`${name} added to bag`);
      };

      window.removeItem = function(id) {
        appState.cart = appState.cart.filter(item => item.id !== id);
        renderCart();
      };

      window.changeQty = function(id, delta) {
        const item = appState.cart.find(i => i.id === id);
        if (item) {
          item.qty += delta;
          if (item.qty <= 0) window.removeItem(id);
          else renderCart();
        }
      };

      function renderCart() {
        const itemsContainer = document.getElementById('cart-items');
        const footer = document.getElementById('cart-footer');
        const countBadge = document.getElementById('cart-count');
        const totalEl = document.getElementById('cart-total');

        const totalQty = appState.cart.reduce((s, i) => s + i.qty, 0);
        const totalPrice = appState.cart.reduce((s, i) => s + (i.price * i.qty), 0);
        
        countBadge.textContent = totalQty;
        totalEl.textContent = `$${totalPrice.toFixed(2)}`;

        if (appState.cart.length === 0) {
          itemsContainer.innerHTML = `
            <div class="flex flex-col items-center justify-center h-full text-brandDark/30">
              <i data-lucide="shopping-bag" class="w-12 h-12 mb-4 opacity-20"></i>
              <p class="font-medium">Your bag is empty</p>
              <button onclick="window.toggleCart()" class="mt-6 text-accent font-bold uppercase tracking-widest text-xs border-b-2 border-accent">Start Shopping</button>
            </div>
          `;
          footer.classList.add('hidden');
        } else {
          footer.classList.remove('hidden');
          itemsContainer.innerHTML = appState.cart.map(item => `
            <div class="flex items-center gap-4">
              <div class="w-20 h-24 bg-brandDark/5 flex-shrink-0 overflow-hidden">
                <img src="${getProductImg(item.id)}" class="w-full h-full object-cover">
              </div>
              <div class="flex-1">
                <div class="flex justify-between items-start mb-1">
                  <h4 class="font-heading font-bold">${item.name}</h4>
                  <button onclick="window.removeItem(${item.id})" class="text-brandDark/30 hover:text-brandDark"><i data-lucide="trash-2" class="w-4 h-4"></i></button>
                </div>
                <p class="text-xs text-brandDark/50 mb-3">$${item.price.toFixed(2)}</p>
                <div class="flex items-center gap-3">
                  <button onclick="window.changeQty(${item.id}, -1)" class="w-6 h-6 border border-brandDark/10 flex items-center justify-center hover:bg-brandDark/5">-</button>
                  <span class="text-sm font-medium">${item.qty}</span>
                  <button onclick="window.changeQty(${item.id}, 1)" class="w-6 h-6 border border-brandDark/10 flex items-center justify-center hover:bg-brandDark/5">+</button>
                </div>
              </div>
            </div>
          `).join('');
        }
        lucide.createIcons();
      }

      function getProductImg(id) {
        const urls = {
          1: 'https://images.pexels.com/photos/3270223/pexels-photo-3270223.jpeg?auto=compress&cs=tinysrgb&w=200',
          2: 'https://images.pexels.com/photos/6707628/pexels-photo-6707628.jpeg?auto=compress&cs=tinysrgb&w=200',
          3: 'https://images.pexels.com/photos/3270224/pexels-photo-3270224.jpeg?auto=compress&cs=tinysrgb&w=200',
          4: 'https://images.pexels.com/photos/9898025/pexels-photo-9898025.jpeg?auto=compress&cs=tinysrgb&w=200',
          5: 'https://images.pexels.com/photos/6690226/pexels-photo-6690226.jpeg?auto=compress&cs=tinysrgb&w=200',
          6: 'https://images.pexels.com/photos/10103767/pexels-photo-10103767.jpeg?auto=compress&cs=tinysrgb&w=200'
        };
        return urls[id];
      }

      window.checkout = function() {
        const totalPrice = appState.cart.reduce((s, i) => s + (i.price * i.qty), 0);
        window.open(`https://paypal.me/bloomandco/${totalPrice.toFixed(2)}`, '_blank', 'noopener,noreferrer');
      };

      window.showToast = function(msg) {
        const toast = document.getElementById('toast');
        document.getElementById('toast-msg').textContent = msg;
        toast.classList.add('show');
        setTimeout(() => toast.classList.remove('show'), 3000);
      };

      // 3. Legal Modals
      window.openLegal = function(title, body) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = body;
        document.getElementById('legalModal').style.display = 'flex';
        document.body.style.overflow = 'hidden';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
        document.body.style.overflow = 'auto';
      };

      // 4. Initialization
      document.getElementById('year').textContent = new Date().getFullYear();

      const io = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
            io.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });

      document.querySelectorAll('.reveal').forEach(el => io.observe(el));
      
      // Safety net for reveal animations
      setTimeout(() => {
        document.querySelectorAll('.reveal').forEach(el => {
          el.style.opacity = '1';
          el.style.transform = 'none';
        });
      }, 1500);

      lucide.createIcons();

      // Smooth scroll override for sticky nav
      document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
          e.preventDefault();
          const targetId = this.getAttribute('href');
          if (targetId === '#') return;
          const target = document.querySelector(targetId);
          if (target) {
            const offset = 80;
            const bodyRect = document.body.getBoundingClientRect().top;
            const elementRect = target.getBoundingClientRect().top;
            const elementPosition = elementRect - bodyRect;
            const offsetPosition = elementPosition - offset;

            window.scrollTo({
              top: offsetPosition,
              behavior: 'smooth'
            });
          }
        });
      });

    })();

        if (appState.cart.length === 0) {
          itemsContainer.innerHTML = `
            <div class="flex flex-col items-center justify-center h-full text-brandDark/30">
              <i data-lucide="shopping-bag" class="w-12 h-12 mb-4 opacity-20"></i>
              <p class="font-medium">Your bag is empty</p>
              <button onclick="window.toggleCart()" class="mt-6 text-accent font-bold uppercase tracking-widest text-xs border-b-2 border-accent">Start Shopping</button>
            </div>
          `;
          footer.classList.add('hidden');
        } else {
          footer.classList.remove('hidden');
          itemsContainer.innerHTML = appState.cart.map(item => {
            const product = appState.products.find(p => p.id === item.id);
            return `
            <div class="flex items-center gap-4">
              <div class="w-20 h-24 bg-brandDark/5 flex-shrink-0 overflow-hidden">
                <img src="${product.img}" class="w-full h-full object-cover">
              </div>
              <div class="flex-1">
                <div class="flex justify-between items-start mb-1">
                  <h4 class="font-heading font-bold">${item.name}</h4>
                  <button onclick="window.removeItem(${item.id})" class="text-brandDark/30 hover:text-brandDark"><i data-lucide="trash-2" class="w-4 h-4"></i></button>
                </div>
                <p class="text-xs text-brandDark/50 mb-3">$${item.price.toFixed(2)}</p>
                <div class="flex items-center gap-3">
                  <button onclick="window.changeQty(${item.id}, -1)" class="w-6 h-6 border border-brandDark/10 flex items-center justify-center hover:bg-brandDark/5">-</button>
                  <span class="text-sm font-medium">${item.qty}</span>
                  <button onclick="window.changeQty(${item.id}, 1)" class="w-6 h-6 border border-brandDark/10 flex items-center justify-center hover:bg-brandDark/5">+</button>
                </div>
              </div>
            </div>
          `}).join('');
        }
        lucide.createIcons();
      }

      window.checkout = function() {
        const totalPrice = appState.cart.reduce((s, i) => s + (i.price * i.qty), 0);
        window.open(`https://paypal.me/bloomandco/${totalPrice.toFixed(2)}`, '_blank', 'noopener,noreferrer');
      };

      window.showToast = function(msg) {
        const toast = document.getElementById('toast');
        const msgEl = document.getElementById('toast-msg');
        if (msgEl) msgEl.textContent = msg;
        toast.classList.add('show');
        setTimeout(() => toast.classList.remove('show'), 3000);
      };

      window.openLegal = function(title, body) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = body;
        document.getElementById('legalModal').style.display = 'flex';
        document.body.style.overflow = 'hidden';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
        document.body.style.overflow = 'auto';
      };

      document.getElementById('year').textContent = new Date().getFullYear();

      const io = new IntersectionObserver(entries => {
        entries.forEach(entry => {
          if (entry.isIntersecting) {
            entry.target.classList.add('visible');
            io.unobserve(entry.target);
          }
        });
      }, { threshold: 0.1 });

      document.querySelectorAll('.reveal').forEach(el => io.observe(el));
      
      lucide.createIcons();
    })();
  </script>
