JavaScriptconst axios = require('axios');
const { GoogleGenerativeAI } = require('@google/generative-ai');

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);

app.post('/chat-verdict', async (req, res) => {
  const { imageBase64, chatQuery, selectedItem } = req.body;

  try {
    const [preResult, searchResponse] = await Promise.all([
      (async () => {
        const prePrompt = `Quick sentiment for ${selectedItem} or query "${chatQuery}" from image: Positive (gold), Warning (red), Safe (green). Output only color.`;
        const preParts = [{ text: prePrompt }, { inlineData: { data: imageBase64, mimeType: "image/png" } }];
        const preModel = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });
        const result = await preModel.generateContent(preParts);
        let color = result.response.text().trim().toLowerCase();
        if (color.includes('positive')) return 'gold';
        if (color.includes('warning')) return 'red';
        return 'green';
      })(),
      (async () => {
        try {
          const searchQuery = `${selectedItem || chatQuery} price on Walmart Target eBay AliExpress news reviews`;
          const response = await axios.get(`https://google.serper.dev/search`, {
            params: { q: searchQuery },
            headers: { 'X-API-KEY': process.env.SERPER_API_KEY }
          });
          return response.data.organic.map(r => `${r.title}: ${r.snippet}`).join('\n');
        } catch (err) {
          return 'No live search - using image/chat analysis.';
        }
      })()
    ]);

    const systemPrompt = `You are the Universal Decision Engine for "The Verdict". Use image, selected item: ${selectedItem}, chat query: "${chatQuery}".
Step 1: Detect context (shopping if prices/products, stocks if charts/RSI, menu if food, etc.).
Step 2: Analyze: Stats from image/query (e.g., RSI/prices).
Step 3: Price crush/alternatives (adapt): Use search/news: ${searchResponse} for comparisons/savings/dupe. For stocks: ETFs/alts with predictions (e.g., "70% chance of 10% gain in 30 days from trends").
Step 4: Insider secret: 1 pro, 1 con from knowledge/reviews/news + bias check/self-reflection (e.g., "Checked for fake reviews/overhype - genuine?").
Step 5: Quality audit: Durability/value/volatility (adapt) + "what if" simulation (e.g., "If wait, save $X with 60% probability from history").
Step 6: Confidence score: Calculate 0-100% dynamically (high if data aligns/low volatility; e.g., RSI>80 lowers 20%, strong reviews adds 10%, bias reduces 15%, multi-source adds 5% - self-reflect/explain calculation reason).
Step 7: Verdict: YES/BUY, NO/AVOID, HOLD/WAIT - with 1-sentence reason (adapt to query like 'sell gold' = risk-focused with alts/predictions).
Output JSON:
{
  "verdict": "YES / BUY",
  "header": "⚖️ THE VERDICT: YES / BUY",
  "insiderSecret": "Pro: Text. Con: Text (with bias self-reflection)",
  "priceCrush": "Text with alts/predictions",
  "qualityAudit": "Text with simulation",
  "confidenceScore": "98% CERTAINTY (self-reflection reason for calculation)",
  "action": "ADD TO CART ➡️",
  "glowColor": "${preResult}"
}`;

    const parts = [{ text: systemPrompt }, { inlineData: { data: imageBase64, mimeType: "image/png" } }];
    const model = genAI.getGenerativeModel({ model: "gemini-1.5-pro" });
    const result = await model.generateContent(parts);
    const verdictJson = JSON.parse(result.response.text().replace(/```json|```/g, ''));

    res.json(verdictJson);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

4. Extension Popup.js (Full Logic)
JavaScript// Chat send
async function sendChat() {
  const chatQuery = getChatInput();
  const imageBase64 = getSnapBase64();
  const selectedItem = getSelectedItem();

  const response = await fetch(`${REPLIT_URL}/chat`, {
    method: 'POST',
    body: JSON.stringify({ imageBase64, chatQuery, selectedItem })
  });
  const data = await response.json();
  appendToChat(data.answer);

  // Show button inline if decisional
  if (chatQuery.toLowerCase().includes('best') || chatQuery.includes('buy') || chatQuery.includes('sell') || chatQuery.includes('verdict')) {
    appendToChat('<button id="verdictBtn" class="verdict-button">⚖️ The Verdict</button>');
    document.getElementById('verdictBtn').addEventListener('click', getVerdict);
  }
}

// getVerdict
async function getVerdict() {
  const imageBase64 = getSnapBase64();
  const chatQuery = getLastChatQuery();
  const selectedItem = getSelectedItem();

  const response = await fetch(`${REPLIT_URL}/chat-verdict`, {
    method: 'POST',
    body: JSON.stringify({ imageBase64, chatQuery, selectedItem })
  });
  const data = await response.json();

  // Glow button
  const btn = document.getElementById('verdictBtn');
  btn.classList.add(data.glowColor);

  // Haptic
  if (navigator.vibrate) navigator.vibrate([200, 100, 200]);

  // Render card as overlay
  const card = document.getElementById('verdictCard');
  card.innerHTML = `
    <div class="verdict-header">${data.header}</div>
    <div class="verdict-icon">⚖️</div>
    <p><strong>Insider Secret:</strong> ${data.insiderSecret}</p>
    <p><strong>Price Crush:</strong> ${data.priceCrush}</p>
    <p><strong>Quality Audit:</strong> ${data.qualityAudit}</p>
    <p><strong>Confidence Score:</strong> ${data.confidenceScore}</p>
    <button>${data.action}</button>
    <button id="closeVerdict" class="close-btn">Close X</button>
  `;
  card.style.display = 'block';

  // Close
  document.getElementById('closeVerdict').addEventListener('click', () => {
    card.style.display = 'none';
  });
}
