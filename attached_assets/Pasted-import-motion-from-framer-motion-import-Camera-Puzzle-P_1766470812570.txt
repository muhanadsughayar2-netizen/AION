const FloatingParticle = ({ delay, duration, x, y, size, type }: { delay: number; duration: number; x: number; y: number; size: number; type: 'star' | 'circle' | 'sparkle' }) => {
  const icons = {
    star: "⭐",
    circle: "✨",
    sparkle: "🌟"
  };
  
  return (
    <motion.div
      className="absolute text-yellow-400 pointer-events-none select-none"
      style={{ left: `${x}%`, top: `${y}%`, fontSize: size }}
      initial={{ opacity: 0, scale: 0, rotate: 0 }}
      animate={{ 
        opacity: [0, 1, 1, 0],
        scale: [0, 1.2, 1, 0.8],
        rotate: [0, 180, 360],
        y: [0, -30, -60, -90]
      }}
      transition={{ 
        duration,
        delay,
        repeat: Infinity,
        repeatDelay: Math.random() * 3
      }}
    >
      {icons[type]}
    </motion.div>
  );
};

const GlowOrb = ({ delay, x, y, color }: { delay: number; x: number; y: number; color: string }) => (

      scale: [1, 1.3, 1],
      opacity: [0.1, 0.25, 0.1],

      y: [0, -20, 0]
    }}
    transition={{ 
      duration: 8,
      delay,
      repeat: Infinity,
      ease: "easeInOut"
    }}
  />
);
