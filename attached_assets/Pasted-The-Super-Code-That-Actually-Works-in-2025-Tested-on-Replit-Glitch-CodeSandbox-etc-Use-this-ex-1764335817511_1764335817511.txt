function captureScreenAsImage() {
  // This runs INSIDE the actual webpage (bypasses most restrictions)
  html2canvas(document.body, {
    useCORS: true,
    allowTaint: true,
    scrollY: -window.scrollY,
    scrollX: -window.scrollX,
    scale: 2,                    // higher quality
    logging: false,
    backgroundColor: null
  }).then(canvas => {
    const dataURL = canvas.toDataURL("image/png");
    
    // Send the perfect screenshot back to popup/background
    chrome.runtime.sendMessage({
      type: "SNAP_DONE",
      image: dataURL,
      title: document.title,
      url: location.href
    });
  });
}
