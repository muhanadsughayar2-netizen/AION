2. The Fixed chat.js
This script uses the incremental parsing model. It removes block-wrapper artifacts (like ```html and markdown markers) the exact moment they appear back from the streamed response chunks and updates the iframe's srcdoc dynamically. This is the spinner-proof code that resolves the 30-minute freeze.

JavaScript
document.addEventListener("DOMContentLoaded", () => {
  // Select DOM Elements
  const tabChat = document.getElementById("tabChat");
  const tabPreview = document.getElementById("tabPreview");
  const tabCode = document.getElementById("tabCode");
  
  const chatPanel = document.getElementById("chatPanel");
  const previewFrameContainer = document.getElementById("previewFrameContainer");
  const codeViewContainer = document.getElementById("codeViewContainer");
  
  const liveIframe = document.getElementById("liveIframe");
  const spinnerOverlay = document.getElementById("spinnerOverlay");
  const spinnerText = document.getElementById("spinnerText");
  const spinnerPercent = document.getElementById("spinnerPercent");
  
  const promptForm = document.getElementById("promptForm");
  const promptText = document.getElementById("promptText");
  const submitBtn = document.getElementById("submitBtn");
  const messagesArea = document.getElementById("messagesArea");
  
  const statusCircle = document.getElementById("statusCircle");
  const statusIndicator = document.getElementById("statusIndicator");
  const charsCounter = document.getElementById("charsCounter");
  const codeDisplay = document.getElementById("codeDisplay");
  const copyCodeBtn = document.getElementById("copyCodeBtn");
  const apiKeyInput = document.getElementById("apiKeyInput");

  let accumulatedResponseText = "";
  let streamActive = false;

  // Tab View Routing Controller
  function switchTab(view) {
    // Normal reset states
    tabChat.classList.remove("bg-white/10", "text-white");
    tabPreview.classList.remove("bg-white/10", "text-white");
    tabCode.classList.remove("bg-white/10", "text-white");
    
    // Default hiding panels
    previewFrameContainer.classList.add("opacity-20 flex");
    codeViewContainer.classList.add("hidden");
    codeViewContainer.classList.remove("flex");

    if (view === "chat") {
      tabChat.classList.add("bg-white/10", "text-white");
      previewFrameContainer.classList.remove("opacity-20");
    } else if (view === "preview") {
      tabPreview.classList.add("bg-white/10", "text-white");
      previewFrameContainer.classList.remove("opacity-20");
    } else if (view === "code") {
      tabCode.classList.add("bg-white/10", "text-white");
      previewFrameContainer.classList.add("hidden");
      codeViewContainer.classList.remove("hidden");
      codeViewContainer.classList.add("flex");
    }
  }

  tabChat.addEventListener("click", () => switchTab("chat"));
  tabPreview.addEventListener("click", () => switchTab("preview"));
  tabCode.addEventListener("click", () => switchTab("code"));

  // Copy code utility trigger
  copyCodeBtn.addEventListener("click", () => {
    navigator.clipboard.writeText(codeDisplay.value);
    copyCodeBtn.textContent = "COPIED!";
    setTimeout(() => { copyCodeBtn.textContent = "Copy to Clipboard"; }, 1500);
  });

  // Real-time Stream Parser System (Removes Markdown Wrappers and updates iframe progressive payload)
  function parseAndRenderProgressive(rawText) {
    if (!rawText) return;

    // Remove markdown block openers and closers
    let cleaned = rawText.trim();
    
    // Strip trailing or leading HTML markdown codeblock characters
    if (cleaned.startsWith("```html")) {
      cleaned = cleaned.slice(7);
    } else if (cleaned.startsWith("```")) {
      cleaned = cleaned.slice(3);
    }

    if (cleaned.endsWith("```")) {
      cleaned = cleaned.slice(0, -3);
    }

    cleaned = cleaned.trim();

    // Display inside the code window live
    codeDisplay.value = cleaned;

    // Inject directly to iframe viewport sandboxed environment
    liveIframe.srcdoc = cleaned;
    charsCounter.textContent = `${rawText.length} bytes`;
  }

  // Handle Form Submission
  promptForm.addEventListener("submit", async (e) => {
    e.preventDefault();
    const prompt = promptText.value.trim();
    if (!prompt || streamActive) return;

    // Initialize UI states
    streamActive = true;
    accumulatedResponseText = "";
    submitBtn.disabled = true;
    promptText.disabled = true;

    statusCircle.className = "w-1.5 h-1.5 rounded-full bg-amber-500 animate-pulse";
    statusIndicator.childNodes[0].nodeValue = " Workspace: Unifying Layers...";

    // Activate the spinner with progressive feedback percentages
    spinnerOverlay.classList.remove("opacity-0");
    spinnerOverlay.classList.add("opacity-100");
    spinnerText.textContent = "Writing HTML Scaffolds...";
    spinnerPercent.textContent = "12% Completed";

    // Append standard user prompt visualization
    const userMsg = document.createElement("div");
    userMsg.className = "bg-white/5 p-3 rounded-xl border border-white/5 text-zinc-300";
    userMsg.textContent = `Prompt: "${prompt}"`;
    messagesArea.appendChild(userMsg);
    messagesArea.scrollTop = messagesArea.scrollHeight;

    // Check custom credentials override
    const localApiKey = apiKeyInput.value.trim();

    try {
      // Simulate/Trigger API Stream (Adjust path according to your extension endpoint)
      const response = await fetch("/api/build", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ 
          prompt: prompt, 
          stage: "classic", // OneShot generation with complete styling in place
          customKey: localApiKey || undefined 
        })
      });

      const body = await response.json();
      if (!body.success) throw new Error(body.error || "Workspace Compiler generated fault state.");

      let rawPayload = body.code || "";
      
      // Spinner percentage acceleration animations
      spinnerText.textContent = "Applying style skins and scripts...";
      spinnerPercent.textContent = "68% Completed";

      // Execute dynamic compilation rendering into Frame
      setTimeout(() => {
        parseAndRenderProgressive(rawPayload);

        // Terminate Loader safely
        spinnerOverlay.classList.remove("opacity-100");
        spinnerOverlay.classList.add("opacity-0");
        
        statusCircle.className = "w-1.5 h-1.5 rounded-full bg-emerald-500";
        statusIndicator.childNodes[0].nodeValue = " Workspace: Code Active";
        
        switchTab("preview"); // Push user cleanly to final live display
      }, 700);

    } catch (err) {
      console.error(err);
      const errMsg = document.createElement("div");
      errMsg.className = "bg-red-500/15 border border-red-500/30 p-3 rounded-xl text-red-400 font-semibold";
      errMsg.textContent = `Engine Error: Could not connect to the compilation service. Please check your network or custom credentials configuration. Details: ${err.message}`;
      messagesArea.appendChild(errMsg);
      messagesArea.scrollTop = messagesArea.scrollHeight;

      spinnerOverlay.classList.remove("opacity-100");
      spinnerOverlay.classList.add("opacity-0");
      statusCircle.className = "w-1.5 h-1.5 rounded-full bg-rose-500";
      statusIndicator.childNodes[0].nodeValue = " Workspace: Build Failed";
    } finally {
      streamActive = false;
      submitBtn.disabled = false;
      promptText.disabled = false;
      promptText.value = "";
    }
  });
});
