JavaScript// FINAL & PERFECT EDIT BUTTON — SHOWS ALL 5 PAGES WHEN YOU CLICK THE PEN
document.querySelectorAll('.thumb-box').forEach((box, thumbIndex) => {
  const editBtn = box.querySelector('.edit-btn'); // your pen icon
  if (!editBtn) return;

  editBtn.onclick = async (e) => {
    e.stopPropagation();

    // This thumbnail = 5 real pages → calculate which ones
    const startPage = thumbIndex * 5;
    const groupPages = allCapturedPages.slice(startPage, startPage + 5);

    if (groupPages.length === 0) return;

    // Load all 5 images
    const images = [];
    for (const src of groupPages) {
      const img = new Image();
      img.src = src;
      await new Promise(r => img.onload = r);
      images.push(img);
    }

    // Stitch them vertically into ONE tall image
    const canvas = document.createElement('canvas');
    canvas.width = images[0].width;
    canvas.height = images.reduce((sum, img) => sum + img.height, 0);

    const ctx = canvas.getContext('2d', { willReadFrequently: true });
    let y = 0;
    images.forEach(img => {
      ctx.drawImage(img, 0, y);
      y += img.height;
    });

    // Open the editor with the full 5-page image
    openAnnotationEditor(canvas.toDataURL('image/png'));
  };
});
