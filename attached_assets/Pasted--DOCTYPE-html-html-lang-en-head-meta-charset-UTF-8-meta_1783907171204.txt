              <button onclick="window.stopBroadcast()" class="px-6 py-2 bg-slate-700 text-white rounded-full font-bold">Stop</button>
            </div>
          </div>
        `;
        lucide.createIcons();
      }

      window.playBroadcast = function() {
        const script = COURSE.lessons[currentLessonIdx].broadcast;
        if (bcLine >= script.length) bcLine = 0;
        
        function playNext() {
          if (bcLine >= script.length) return;
          const line = script[bcLine];
          
          // Update UI
          Object.keys(VOICES).forEach(name => {
            document.getElementById(`char-${name}`).classList.toggle('active', name === line.speaker);
          });
          
          const transcript = document.getElementById('transcript');
          if (bcLine === 0) transcript.innerHTML = '';
          transcript.innerHTML += `<p class="mb-2"><span style="color: ${VOICES[line.speaker].color}">● ${line.speaker}:</span> ${line.text}</p>`;
          transcript.scrollTop = transcript.scrollHeight;

          const u = new SpeechSynthesisUtterance(line.text);
          u.onend = () => { bcLine++; playNext(); };
          window.speechSynthesis.speak(u);
        }
        window.speechSynthesis.cancel();
        playNext();
      };

      window.stopBroadcast = function() {
        window.speechSynthesis.cancel();
        bcLine = 0;
        renderBroadcast();
      };

      // Video Logic
      function renderVideo() {
        const container = document.getElementById('tab-content');
        container.innerHTML = `
          <div class="card text-center">
            <canvas id="videoCanvas" width="640" height="360" class="w-full max-w-2xl mx-auto rounded-xl bg-[#0D1117] shadow-2xl mb-6"></canvas>
            <div class="flex justify-center gap-4">
              <button onclick="window.startVideo()" class="px-8 py-3 bg-indigo-600 text-white rounded-xl font-bold">▶ Watch Explainer</button>
              <button onclick="window.downloadVideo()" class="px-8 py-3 bg-slate-800 text-white rounded-xl font-bold">⬇ Download .webm</button>
            </div>
          </div>
        `;
      }

      window.startVideo = function() {
        const canvas = document.getElementById('videoCanvas');
        const ctx = canvas.getContext('2d');
        const steps = COURSE.lessons[currentLessonIdx].videoSteps;
        let stepIdx = 0;

        function runStep() {
          if (stepIdx >= steps.length) return;
          const step = steps[stepIdx];
          
          ctx.fillStyle = '#0D1117'; ctx.fillRect(0,0,640,360);
          ctx.textAlign = 'center';

          if (step.type === 'title') {
            ctx.fillStyle = step.color;
            ctx.font = 'bold 40px Inter';
            ctx.fillText(step.emoji + ' ' + step.text, 320, 180);
          } else if (step.type === 'concept') {
            ctx.strokeStyle = step.color; ctx.lineWidth = 2;
            ctx.strokeRect(120, 100, 400, 160);
            ctx.fillStyle = step.color;
            ctx.font = 'bold 32px Inter';
            ctx.fillText(step.emoji + ' ' + step.label, 320, 160);
            ctx.fillStyle = 'white';
            ctx.font = '18px Inter';
            ctx.fillText(step.body, 320, 210);
          } else if (step.type === 'arrow') {
            ctx.strokeStyle = 'white'; ctx.beginPath(); ctx.moveTo(200, 180); ctx.lineTo(440, 180); ctx.stroke();
            ctx.fillStyle = 'white'; ctx.fillText('▶', 440, 186);
          } else if (step.type === 'end') {
            ctx.fillStyle = '#10B981'; ctx.font = 'bold 40px Inter';
            ctx.fillText(step.emoji + ' ' + step.text, 320, 180);
          }

          window.speak(step.label || step.text);
          setTimeout(() => { stepIdx++; runStep(); }, step.duration);
        }
        runStep();
      };

      window.downloadVideo = function() {
        const canvas = document.getElementById('videoCanvas');
        const stream = canvas.captureStream(30);
        const recorder = new MediaRecorder(stream, { mimeType: 'video/webm' });
        const chunks = [];
        recorder.ondataavailable = e => chunks.push(e.data);
        recorder.onstop = () => {
          const blob = new Blob(chunks, { type: 'video/webm' });
          const url = URL.createObjectURL(blob);
          const a = document.createElement('a'); a.href = url; a.download = 'lesson-explainer.webm'; a.click();
        };
        recorder.start();
        window.startVideo();
        setTimeout(() => recorder.stop(), COURSE.lessons[currentLessonIdx].videoSteps.reduce((a,b)=>a+b.duration, 0));
      };

      function updateProgress() {
        const progress = JSON.parse(localStorage.getItem('ap12_progress') || '{}');
        progress[currentLessonIdx] = true;
        localStorage.setItem('ap12_progress', JSON.stringify(progress));
        
        const count = Object.keys(progress).length;
        const total = COURSE.lessons.length;
        const pct = Math.round((count / total) * 100);
        
        document.getElementById('overall-progress-text').innerText = pct + '%';
        document.getElementById('overall-progress-bar').style.width = pct + '%';
      }

      // Init
      window.switchLesson(0);
      updateProgress();
    })();
