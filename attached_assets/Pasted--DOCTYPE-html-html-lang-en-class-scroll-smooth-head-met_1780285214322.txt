  <script>
    (function() {
      // Initialize Lucide
      lucide.createIcons();

      // Intersection Observer for Reveal
      const _io = new IntersectionObserver(entries => {
        entries.forEach(e => {
          if (e.isIntersecting) {
            e.target.classList.add('visible');
            _io.unobserve(e.target);
          }
        });
      }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });
      
      document.querySelectorAll('.reveal').forEach(el => _io.observe(el));
      
      // Safety net for animations
      setTimeout(() => {
        document.querySelectorAll('.reveal').forEach(el => {
          el.style.opacity = '1';
          el.style.transform = 'none';
        });
      }, 1500);

      // Tabs Logic
      window.showTab = function(id, btn) {
        document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));
        document.querySelectorAll('.tab-btn').forEach(b => {
          b.classList.remove('active', 'bg-forest', 'text-cream');
          b.classList.add('border-forest/10');
        });
        
        const panel = document.getElementById(id);
        if (panel) panel.classList.add('active');
        if (btn) {
          btn.classList.add('active', 'bg-forest', 'text-cream');
          btn.classList.remove('border-forest/10');
        }

        // Trigger ring animation if on diet tab
        if (id === 'diet') {
          const ring = document.getElementById('macroRing');
          ring.style.strokeDashoffset = 691;
          setTimeout(() => ring.style.strokeDashoffset = 200, 100);
        }
      };

      // Modal Logic
      window.openLegal = function(title, body) {
        document.getElementById('legalTitle').textContent = title;
        document.getElementById('legalBody').innerHTML = body;
        document.getElementById('legalModal').style.display = 'flex';
        document.body.style.overflow = 'hidden';
      };

      window.closeLegal = function() {
        document.getElementById('legalModal').style.display = 'none';
        document.body.style.overflow = 'auto';
      };

      // Parallax Effect
      window.addEventListener('scroll', () => {
        const scrolled = window.pageYOffset;
        const parallax = document.querySelector('.parallax-bg');
        if (parallax) {
          parallax.style.transform = `translateY(${scrolled * 0.1}px)`;
        }
      });

      // Smooth Scroll with Offset
      document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
          e.preventDefault();
          const targetId = this.getAttribute('href');
          if (targetId === '#') return;
          const target = document.querySelector(targetId);
          if (target) {
            window.scrollTo({
              top: target.getBoundingClientRect().top + window.pageYOffset - 80,
              behavior: 'smooth'
            });
          }
        });
      });

      // Initial tab state
      const initialBtn = document.querySelector('.tab-btn.active');
      if (initialBtn) {
        initialBtn.classList.add('bg-forest', 'text-cream');
        initialBtn.classList.remove('border-forest/10');
      }

    })();
  </script>

            e.target.classList.add('visible');

          }
        });
      }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

          b.classList.add('border-forest/10');

        const panel = document.getElementById(id);
        if (panel) panel.classList.add('active');
