3. popup.js (The Logic)
This handles the "flow" by saving the key and instantly showing the upgrade options.

JavaScript

const apiKeyInput = document.getElementById('apiKey');
const proDashboard = document.getElementById('proDashboard');
const freeLink = document.getElementById('freeLink');

// This function unlocks the Pro UI
function unlockProFeatures() {
  freeLink.style.display = 'none';
  proDashboard.classList.remove('hidden');
}

// Watch for the user to paste their key
apiKeyInput.addEventListener('input', (event) => {
  const value = event.target.value.trim();
  
  if (value.length > 20) {
    // Save to Chrome Storage
    chrome.storage.local.set({ 'gemini_api_key': value }, () => {
      unlockProFeatures();
    });
  }
});

// Check if a key already exists when the user opens the extension
chrome.storage.local.get('gemini_api_key', (result) => {
  if (result.gemini_api_key) {
    apiKeyInput.value = result.gemini_api_key;
    unlockProFeatures();
  }
});
