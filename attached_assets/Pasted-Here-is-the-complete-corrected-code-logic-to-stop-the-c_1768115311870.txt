async function executePlan(plan) {
    // FIX: Initialize controller early to avoid 'signal' errors
    const controller = new AbortController(); 
    const signal = controller.signal;
    
    try {
        for (const step of plan.steps) {
            console.log(`Executing: ${step.description}`);
            
            // --- ACTION: CLICK (Smart Search) ---
            if (step.action === 'click') {
                await chrome.scripting.executeScript({
                    target: { tabId: targetTabId },
                    func: (selector, btnText) => {
                        // Plan A: Try the precise CSS Selector
                        let el = document.querySelector(selector);
                        
                        // Plan B: If Selector fails, find by TEXT (e.g., "1y", "3m")
                        if (!el && btnText) {
                            // broaden search to common clickable elements
                            const buttons = Array.from(document.querySelectorAll('button, a, span, div[role="button"]'));
                            el = buttons.find(b => b.innerText.trim().toLowerCase() === btnText.toLowerCase());
                        }
                        
                        if (el) {
                            // Ensure element is visible
                            el.scrollIntoView({ behavior: 'smooth', block: 'center' });
                            // rapid-fire click for reliability
                            el.click();
                            return true;
                        }
                        return false;
                    },
                    args: [step.selector, step.text || step.description] // Pass text description as backup
                });
            }

            // --- ACTION: WAIT (Essential for loading charts) ---
            if (step.action === 'wait') {
                console.log(`Waiting ${step.seconds || 2} seconds...`);
                await new Promise(resolve => setTimeout(resolve, (step.seconds || 2) * 1000));
            }

            // --- ACTION: SCREENSHOT ---
            if (step.action === 'screenshot') {
                // Ensure captureTab uses your existing tab capture logic
                const imageData = await captureTab(targetTabId);
                if (imageData) {
                    // Update this to match your existing image saving function (e.g., capturedImages.push)
                    if (typeof saveCapturedImage === 'function') {

                        // Fallback if function name differs
                        console.log("Image captured (saveCapturedImage function missing)");
                        capturedImages.push(imageData); 
                        // Trigger UI update if needed
                    }
                }
            }
        }
    } catch (error) {
        console.error("Agent Execution Error:", error);
    }
}
