B. The Excel "One-Cell" Fix (Add to background.js inside agentExecute)
This uses CDP to physically press "Enter," which is the only way to move the cursor in Excel Online reliably.

if (executeAction === 'selectCellRange') {
  (async () => {
    // 1. Type range into Name Box via content script
    await chrome.tabs.sendMessage(tabId, { action: 'agentExecute', executeAction: 'type', params: { text: params.range, pressEnter: true } });
    
    // 2. CDP Force: Press Enter to commit the selection
    await chrome.debugger.attach({ tabId }, '1.3');
    await chrome.debugger.sendCommand({ tabId }, 'Input.dispatchKeyEvent', { type: 'keyDown', windowsVirtualKeyCode: 13, nativeVirtualKeyCode: 13, key: 'Enter' });
    await chrome.debugger.sendCommand({ tabId }, 'Input.dispatchKeyEvent', { type: 'keyUp', key: 'Enter' });
    await chrome.debugger.detach({ tabId });
    
    sendResponse({ success: true });
  })();
  return true;
}
C. The "Name Box" Finder (Add to content.js switch case)
This allows the agent to find the "jump to cell" box in both Google and Microsoft apps.

case 'selectCellRange':
  const nameBox = document.querySelector('#t-name-box') || // Google Sheets
                  document.querySelector('input[aria-label*="Name Box"]') || // Excel Online
                  document.querySelector('.name-box-container input'); // Generic
  if (nameBox) {
    nameBox.value = params.range;
    nameBox.dispatchEvent(new Event('input', { bubbles: true }));
    nameBox.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', bubbles: true }));
    return { success: true };
  }
  return { success: false, error: "Name box not found" };
Summary of Why This Works
Excel/Sheets Fix: Your current agent types in one cell because it never "commits" the entry. By adding selectCellRange and a physical CDP Enter key press, the browser is forced to move the focus to the next cell or select the whole table area. [1]
Word/Docs Images: By adding insertImage, Gemini knows it needs to use a specific flow (Menu -> Submenu -> URL/Upload) rather than just clicking randomly on the canvas. [2]
