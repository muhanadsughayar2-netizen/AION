<script>
  // Tabs Logic
  function showTab(id, btn) {
    document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));
    document.querySelectorAll('.app-tab').forEach(b => b.classList.remove('active'));
    const panel = document.getElementById(id);
    if (panel) panel.classList.add('active');
    if (btn) btn.classList.add('active');
  }

  // Visualizer Trigger
  function triggerProcess() {
    const btn = document.querySelector('.process-btn');
    btn.textContent = 'GENERATING...';
    btn.style.opacity = '0.7';
    setTimeout(() => {
      btn.textContent = 'SUCCESS: VIEW IN STUDIO';
      btn.style.background = '#22c55e';
      btn.style.opacity = '1';
    }, 2000);
  }

  // Intersection Observer
  const _io = new IntersectionObserver(entries => {
    entries.forEach(e => {
      if (e.isIntersecting) {
        e.target.classList.add('visible');
        _io.unobserve(e.target);
      }
    });
  }, { threshold: 0.1 });

  document.querySelectorAll('.reveal').forEach(el => _io.observe(el));
  
  // Safety fallback
  setTimeout(() => {
    document.querySelectorAll('.reveal').forEach(el => {
      el.style.opacity = '1';
      el.style.transform = 'none';
    });
  }, 1000);

  // Modal Logic
  const modal = document.getElementById('legalModal');
  const modalTitle = document.getElementById('modalTitle');
  const modalBody = document.getElementById('modalBody');

  function openLegal(title, content) {
    modalTitle.textContent = title;
    modalBody.innerHTML = content;
    modal.style.display = 'flex';
  }

  function closeLegal() {
    modal.style.display = 'none';
  }

  window.onclick = function(event) {
    if (event.target == modal) closeLegal();
  }

  // Legal Content Strings
  const privacyContent = `
    <p>Your privacy is paramount at Veoflow. We do not store or transmit your captured pixels to our servers. All image processing occurs within the secure sandbox of your browser using the Google Gemini API directly.</p>
    <p>We do not collect personal identification information. Usage metrics are anonymized and used solely to improve the performance of our local generation models.</p>
    <p>By using the Veoflow extension, you agree to the processing of data as described in this policy.</p>
  `;

  const termsContent = `
    <p>Veoflow provides a browser-based AI generation tool. Users are responsible for the content they capture and generate. You must have the rights to any screen content you capture for processing.</p>
    <p>Generative outputs are subject to the Google Gemini and VEO usage guidelines. Any misuse of the tool for generating harmful, illegal, or deceptive content will result in immediate termination of your Pro license.</p>
    <p>The service is provided "as is" without warranty of any kind.</p>
  `;

  const cookiesContent = `
    <p>We use essential cookies to manage your Chrome Extension authentication and Pro status. We do not use tracking cookies for third-party advertising.</p>
    <p>LocalStorage is used to store your preferred Studio settings and recent prompt history locally on your machine.</p>
  `;
