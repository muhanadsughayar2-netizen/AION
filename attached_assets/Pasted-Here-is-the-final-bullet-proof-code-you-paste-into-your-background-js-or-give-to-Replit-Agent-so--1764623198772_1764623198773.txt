chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.action === "startFullPageInAI") {
    chrome.tabs.query({ active: true, currentWindow: true }, (tabs) => {
      const tabId = tabs[0].id;

      // Force re-inject into every iframe on every capture (this is the magic)
      const injectCaptureScript = () => {
        chrome.scripting.executeScript({
          target: { tabId: tabId, allFrames: true },
          func: runFullPageCapture,
        });
      };

      // Run it now + every time the page changes (Grok/Claude do this constantly)
      injectCaptureScript();
      chrome.tabs.onUpdated.addListener(function listener(tabIdUpdated, info) {
        if (tabIdUpdated === tabId && info.status === "complete") {
          setTimeout(injectCaptureScript, 800);
        }
      });
    });
    sendResponse({ status: "started" });
  }
});

// This function runs inside the page (even inside Grok’s iframes)
function runFullPageCapture() {
  let images = [];
  let scrollPos = 0;
  const viewportH = window.innerHeight;
  const totalH = Math.max(
    document.body.scrollHeight,
    document.body.offsetHeight,
    document.documentElement.scrollHeight
  );

  const captureStep = async () => {
    const dataUrl = await chrome.runtime.sendMessage({ action: "captureVisible" });
    images.push(dataUrl);

    scrollPos += viewportH;
    window.scrollTo(0, scrollPos);

    if (scrollPos < totalH) {
      setTimeout(captureStep, 600);
    } else {
      // All done → send back to background to stitch
      chrome.runtime.sendMessage({
        action: "fullPageDone",
        images: images,
        url: location.href
      });
    }
  };

  captureStep();
}

// Handle the visible-tab capture from background
chrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.action === "captureVisible") {
    chrome.tabs.captureVisibleTab(null, { format: "png" }, (dataUrl) => {
      sendResponse(dataUrl);
    });
    return true; // async response
  }

  if (msg.action === "fullPageDone") {
    // Stitch all images into one canvas and add to your tray
    stitchAndAddToTray(msg.images);
  }
});
