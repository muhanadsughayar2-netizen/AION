To get this working in Replit, you need a piece of code that is "Plug and Play." I have written this as a React component using Tailwind CSS, which is what Replit Agents and modern web apps use to make things look clean and "app-like."

Copy the code below. You can tell the Replit Agent: "Replace my current chat bubble logic with this Verdict Card component."

The Full Replit Code (React + Tailwind)
import React from 'react';

const VerdictCard = ({ data }) => {
  // Logic to handle colors based on the "Final Call"
  const getCallStyle = (call) => {
    switch (call.toUpperCase()) {
      case 'BUY': return 'text-green-600 font-black';
      case 'SKIP': return 'text-red-500 font-black';
      case 'CONSIDER': return 'text-orange-500 font-black';
      case 'WAIT': return 'text-blue-500 font-black';
      default: return 'text-slate-600 font-black';
    }
  };

  return (
    <div className="max-w-md mx-auto my-4 bg-white border-l-8 border-blue-600 rounded-xl shadow-lg overflow-hidden font-sans">
      {/* Header Section */}
      <div className="flex justify-between items-center bg-slate-50 px-4 py-3 border-b border-slate-100">
        <h3 className="text-sm font-extrabold text-slate-800 tracking-tight">
          ⚖️ THE VERDICT CARD
        </h3>
        <div className="flex items-center bg-orange-100 px-2 py-1 rounded-full">
          <span className="text-[10px] font-bold text-orange-700 uppercase tracking-widest">
            Vibe: {data.vibe || "🔥 Hot Deals"}
          </span>
        </div>
      </div>

      {/* Product List Section */}
      <div className="p-0">
        {data.items.map((item, index) => (
          <div 
            key={index} 
            className={`flex items-center justify-between px-4 py-3 ${
              index !== data.items.length - 1 ? 'border-b border-slate-50' : ''
            } hover:bg-slate-50 transition-colors cursor-pointer`}
          >
            {/* Left Side: Info */}
            <div className="flex flex-col">
              <span className="text-sm font-bold text-slate-800 leading-tight">
                {item.name}
              </span>
              <div className="flex items-center gap-1 mt-0.5">
                <span className="text-xs">{item.emoji}</span>
                <span className="text-[11px] font-bold text-slate-500 uppercase">
                  {item.tag}
                </span>
                <span className="text-[11px] text-slate-400">• {item.context}</span>
              </div>
            </div>

            {/* Right Side: Action */}
            <div className="flex flex-col items-end">
              <span className={`text-sm tracking-tighter ${getCallStyle(item.call)}`}>
                {item.call}
              </span>
            </div>
          </div>
        ))}
      </div>

      {/* Footer Section */}
      <div className="bg-slate-50 px-4 py-2 text-center">
        <p className="text-[10px] text-slate-400 italic">
          Ask me <span className="font-bold">"Why?"</span> for any item to see the market data
        </p>
      </div>
    </div>
  );
};

export default VerdictCard;
