    const model = "gemini-3-flash-preview";
    const genAI = new GoogleGenerativeAI(apiKey); // Assuming you have your API key set

    const generativeModel = genAI.getGenerativeModel({ model });

    const parts = [
        { text: prompt },
        { image: { url: imageURL, mimeType: "image/jpeg" } } // Adjust mimeType as needed
    ];

    try {
        const result = await generativeModel.generateContent({ contents: parts });
        const response = await result.response;
        console.log(response.text());

        return "Error analyzing image.";
    }
}
