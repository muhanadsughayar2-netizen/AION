JavaScript// Done Cropping - SMART behavior for full-page
document.getElementById('doneCropBtn').addEventListener('click', () => {
  if (!cropRect || cropRect.width < 50 || cropRect.height < 50) {
    updateStatus('Crop area too small (min 50x50)');
    return;
  }

  pushHistory(); // Save undo state

  const { x, y, width, height } = cropRect;

  if (isFullPageMode && pages.length > 1) {
    // === FULL-PAGE: Apply only left/right crop to ALL pages ===
    updateStatus(`Removing sidebars from all ${pages.length} pages...`);

    const sideCropX = x;
    const sideCropWidth = width;

    for (let i = 0; i < pages.length; i++) {
      if (!pageImages[i]) continue;

      const img = pageImages[i];
      const fullHeight = img.height;

      const tempCanvas = document.createElement('canvas');
      tempCanvas.width = sideCropWidth;
      tempCanvas.height = fullHeight;
      const tctx = tempCanvas.getContext('2d');
      tctx.drawImage(img, sideCropX, 0, sideCropWidth, fullHeight, 0, 0, sideCropWidth, fullHeight);

      const newImg = new Image();
      newImg.onload = () => {
        pageImages[i] = newImg;
        pageOriginalImages[i] = tctx.getImageData(0, 0, sideCropWidth, fullHeight);

        // Shift annotations horizontally only (preserve vertical position)
        const anns = pageAnnotations[i] || [];
        anns.forEach(ann => {
          ann.x -= sideCropX;
        });
        pageAnnotations[i] = anns;
      };
      newImg.src = tempCanvas.toDataURL();
    }

    // Apply to current page (keep user's top/bottom if adjusted)
    applyCropToCurrentPage(sideCropX, 0, sideCropWidth, canvas.height);

    updateStatus(`Sidebars removed from all ${pages.length} pages! Top/bottom preserved.`);
  } else {
    // Single image: full crop (including top/bottom)
    applyCropToCurrentPage(x, y, width, height);
    updateStatus('Crop applied.');
  }

  exitCropMode();
});
