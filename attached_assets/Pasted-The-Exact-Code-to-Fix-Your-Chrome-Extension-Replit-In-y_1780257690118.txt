In your Chrome Extension's preview-output.js (or where your runScripts function is declared), replace your script loader with this modern, leak-proof version. It isolates executed scripts inside an anonymous scope so const and let can be re-declared infinitely without crashing:

function runScripts(container) {
  try {
    container.querySelectorAll('script').forEach(function(old) {
      try {
        if (!old.textContent.trim() && !old.hasAttribute('src')) return;

        var fresh = document.createElement('script');
        Array.from(old.attributes).forEach(function(a) {
          try { fresh.setAttribute(a.name, a.value); } catch(e) {}
        });

        if (old.textContent.trim()) {
          // SAFEGUARD IIFE: Wrap the inline script so const/let declarations
          // don't leak into the global sandbox window namespace on updates.
          fresh.textContent = ";(function(){\n" + old.textContent + "\n})();";
        }

        if (old.parentNode) {
          old.parentNode.replaceChild(fresh, old);
        }
      } catch(e) {
        console.error('[SnapToAI Sandbox Error]', e);
      }
    });
  } catch(e) {}
}

LEAK-PROOF SCRIPT SCOPING:
To prevent global re-declaration crashes (SyntaxError: Identifier already declared) in Chrome Extension previews, you are strictly forbidden from placing bare "let" and "const" variables at the top level of style or visual script tags.
Always seal your page-logic scripts inside an anonymous IIFE block:

(function() {
  // Your IntersectionObservers and variables go here
  const observer = new IntersectionObserver(...);
