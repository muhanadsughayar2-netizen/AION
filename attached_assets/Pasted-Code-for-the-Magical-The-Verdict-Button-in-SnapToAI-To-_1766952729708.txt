Code for the Magical "The Verdict ⚖️" Button in SnapToAI
To implement this in your Replit (Node.js backend) and Chrome extension, we'll add:

Backend (/verdict endpoint in index.js): Handles AI processing with the universal prompt, pre-scan for glow color (gold/green/red based on sentiment), and returns JSON for the card.
Frontend (extension popup.html, style.css, popup.js): The glowing, pulsing button with animations/haptics (using CSS/JS; haptics via navigator.vibrate if supported). The card renders with visual wow (3D text, icons, ripple).

This eliminates competitors by being screenshot-first, multi-scenario (shopping/stocks/menus via prompt), and visually magical—glowing based on AI pre-scan, animated reveal.
1. Replit Backend Update (Add to index.js)
Add this endpoint. It pre-scans for glow color (quick AI sentiment check), then full verdict. Install axios if not (npm install axios for search).
JavaScriptconst axios = require('axios');
const { GoogleGenerativeAI } = require('@google/generative-ai');

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);

app.post('/verdict', async (req, res) => {
  const { imageBase64, selectedItem, context = 'shopping' } = req.body; // image, item, optional context (shopping/stocks/etc.)

  try {
    // Pre-scan for glow color (quick sentiment)
    const prePrompt = `Quick sentiment for ${selectedItem} in ${context} context from image: Positive (gold), Warning (red), Safe (green). Output only color.`;
    const preParts = [{ text: prePrompt }, { inlineData: { data: imageBase64, mimeType: "image/png" } }];
    const preResult = await genAI.getGenerativeModel({ model: "gemini-1.5-flash" }).generateContent(preParts);
    const glowColor = preResult.response.text().toLowerCase().includes('positive') ? 'gold' : preResult.response.text().toLowerCase().includes('warning') ? 'red' : 'green';

    // Search for comparisons (using Serper for real-time prices/dupe)
    const searchQuery = `${selectedItem} price on Walmart Target eBay AliExpress`;
    const searchResponse = await axios.get(`https://google.serper.dev/search`, {
      params: { q: searchQuery },
      headers: { 'X-API-KEY': process.env.SERPER_API_KEY } // Add Serper key to secrets
    });
    const searchResults = searchResponse.data.organic.map(r => `${r.title}: ${r.snippet}`).join('\n');

    // Full universal prompt
    const systemPrompt = `You are the Universal Decision Engine for "The Verdict". User provides image and selected item: ${selectedItem}. Context: ${context}.
Step 1: Identify (e.g., shopping=products/prices, stocks=charts/trends, menu=choices).
Step 2: Analyze: Name, price/stats from image.
Step 3: Price crush/dupe: Use search results: ${searchResults} to find cheaper on Walmart/Target/eBay/AliExpress. Calculate savings.
Step 4: Insider secret: Pro/con from reviews/knowledge.
Step 5: Quality audit: Durability/accuracy/value.
Step 6: Confidence score: 0-100% based on data.
Step 7: Verdict: YES/BUY, NO/AVOID, HOLD/WAIT - with reason.
Output JSON:
{
  "verdict": "YES / BUY",
  "header": "⚖️ THE VERDICT: YES / BUY",
  "insiderSecret": "Text",
  "priceCrush": "Text",
  "qualityAudit": "Text",
  "confidenceScore": "98% CERTAINTY",
  "action": "ADD TO CART ➡️",
  "glowColor": "${glowColor}"
}`;

    const parts = [{ text: systemPrompt }, { inlineData: { data: imageBase64, mimeType: "image/png" } }];
    const result = await genAI.getGenerativeModel({ model: "gemini-1.5-pro" }).generateContent(parts);
    const verdictJson = JSON.parse(result.response.text());

    res.json(verdictJson);
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

Add to popup.js (after snap/image capture):
JavaScriptdocument.getElementById('verdictBtn').addEventListener('click', async () => {
  const imageBase64 = getImageBase64(); // Your snap function to get base64
  const selectedItem = getSelectedItem(); // From user select or AI identify
  const context = 'shopping'; // Or detect/auto

  // Send to Replit
  const response = await fetch(`${REPLIT_URL}/verdict`, {
    method: 'POST',
    body: JSON.stringify({ imageBase64, selectedItem, context })
  });
  const data = await response.json();

  // Update button glow from pre-scan
  const btn = document.getElementById('verdictBtn');
  btn.classList.add(data.glowColor); // gold/red/green

  // Haptic thud (if supported)
  if (navigator.vibrate) navigator.vibrate(200);

  // Render card with ripple
  const card = document.getElementById('verdictCard');
  card.innerHTML = `
    <div class="verdict-header">${data.header}</div>
    <div class="verdict-icon">⚖️</div>
    <p><strong>Insider Secret:</strong> ${data.insiderSecret}</p>
    <p><strong>Price Crush:</strong> ${data.priceCrush}</p>
    <p><strong>Quality Audit:</strong> ${data.qualityAudit}</p>
    <p><strong>Confidence Score:</strong> ${data.confidenceScore}</p>
    <button>${data.action}</button>
  `;
  card.style.display = 'block';
  // Ripple animation via CSS class
});
