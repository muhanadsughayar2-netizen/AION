Task: Add Lossless PDF Optimization to SnapToAI

Goal: Reduce PDF file size without changing a single pixel of the images (for AI vision compatibility).

Instructions:

Modify popup.html: Add this script tag before the popup.js script tag: <script src="https://unpkg.com/pdf-lib@1.17.1/dist/pdf-lib.min.js"></script>

async function handleExportPDFDirect() {
    console.log("Starting Optimized PDF Export...");
    try {
        // Capture the original bytes from jsPDF (assuming 'doc' is the jsPDF instance)
        const originalBytes = doc.output('arraybuffer');
        
        // Load into pdf-lib for lossless structural compression
        const { PDFDocument } = PDFLib;
        const pdfDoc = await PDFDocument.load(originalBytes);
    // Save with 'useObjectStreams' to shrink the file size losslessly
    const optimizedBytes = await pdfDoc.save({
        useObjectStreams: true,
        addDefaultPage: false,
        updateFieldAppearances: false
    });
    // Safe Download Trigger
    const blob = new Blob([optimizedBytes], { type: 'application/pdf' });
    const downloadUrl = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = downloadUrl;
    link.download = `SnapToAI_Export_${Date.now()}.pdf`;
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    // Delay revocation to prevent download failure
    setTimeout(() => URL.revokeObjectURL(downloadUrl), 10000);
    
    console.log("Lossless PDF optimization complete.");
} catch (error) {
    console.error("Optimization failed, falling back to standard save:", error);
    doc.save('SnapToAI_Standard.pdf');
}
}
