3. Popup.js (Bulb Hover for Tooltip)
JavaScriptdocument.addEventListener('DOMContentLoaded', () => {
  const helpBulb = document.getElementById('helpBulb');
  const helpTooltip = document.getElementById('helpTooltip');
  const upgradeBtn = document.getElementById('upgradeBtn');

  // Bulb hover for tooltip (optional more info)
  helpBulb.addEventListener('mouseover', () => { helpTooltip.style.display = 'block'; });
  helpBulb.addEventListener('mouseout', () => { helpTooltip.style.display = 'none'; });
  helpBulb.addEventListener('click', () => { helpTooltip.style.display = helpTooltip.style.display === 'block' ? 'none' : 'block'; });

  // Upgrade button - links to Google Cloud for $300 credits
  upgradeBtn.addEventListener('click', () => { window.open('https://console.cloud.google.com/billing', '_blank'); });
});
