    function showTab(id, btn) {

    /* 3. METABOLIC BIO-VISUALIZER LOGIC */
    function activateBio(name, percent, color) {
      const circle = document.getElementById('progressCircle');
      const text = document.getElementById('percentText');
      const label = document.getElementById('activeFoodName');
      
      // Calculate SVG stroke offset (circumference of r=130 is ~816)
      const circumference = 2 * Math.PI * 130;
      const offset = circumference - (percent / 100) * circumference;
      
      circle.style.stroke = color;
      circle.style.strokeDashoffset = offset;
      
      // Animate percentage text
      let current = 0;
      const duration = 1000;
      const startTime = performance.now();
      
      function updateText(timestamp) {
        const progress = Math.min((timestamp - startTime) / duration, 1);
        const value = Math.floor(progress * percent);
        text.innerText = value + '%';
        if (progress < 1) requestAnimationFrame(updateText);
      }
      
      requestAnimationFrame(updateText);
      
      label.innerText = name;
      label.style.color = color;
    }

    function openLegal(title, html) {
      const modal = document.getElementById('legalModal');
      document.getElementById('legalTitle').innerText = title;
      document.getElementById('legalBody').innerHTML = html;
      modal.style.display = 'flex';
      document.body.style.overflow = 'hidden';
    }

    function closeLegal() {
      document.getElementById('legalModal').style.display = 'none';
      document.body.style.overflow = 'auto';
    }

    // Close modal on outside click
    window.onclick = function(event) {
      const modal = document.getElementById('legalModal');
      if (event.target == modal) closeLegal();
    }

    // Initial state for visualizer
    setTimeout(() => {
      activateBio('Metabolic Optimization', 0, 'var(--accent)');
    }, 500);
  </script>
