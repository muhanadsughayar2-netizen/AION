Modify startFullPageCapture() to detect Google Docs and use the debugger method. Fall back to your existing scrolling method for other sites.
JavaScript// In background.js, replace startFullPageCapture with this:

async function startFullPageCapture() {
  try {
    if (isFullPageCaptureInProgress) {
      return { success: false, error: 'Capture in progress', isExpected: true };
    }

    const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });
    if (!isCapturableUrl(tab.url)) {
      return { success: false, error: 'Cannot capture this page', isExpected: true };
    }

    isFullPageCaptureInProgress = true;

    // Detect Google Docs (URL like docs.google.com/document)
    const isGoogleDocs = tab.url.includes('docs.google.com/document');

    if (isGoogleDocs) {
      // Use debugger API for Google Docs
      const dataUrl = await captureFullPageWithDebugger(tab.id);
      if (!dataUrl) {
        throw new Error('Capture failed');
      }

      // Save as single snap (or chunk if large)
      await chrome.runtime.sendMessage({
        action: 'saveSnap',
        dataUrl: dataUrl,
        isSnip: false,
        chunkInfo: null,
        smartName: 'Google Doc'
      });

      isFullPageCaptureInProgress = false;
      return { success: true };
    } else {
      // Existing scrolling method for other sites
      await chrome.scripting.executeScript({
        target: { tabId: tab.id, allFrames: true },
        files: ['content.js']
      });

      await new Promise(resolve => setTimeout(resolve, 100));

      await chrome.tabs.sendMessage(tab.id, {
        action: 'startFullPageScroll',
        tabId: tab.id
      });

      return { success: true };
    }
  } catch (error) {
    console.log('[SnapToAI] Capture:', error);
    isFullPageCaptureInProgress = false;
    return { success: false, error: 'Capture failed', isExpected: true };
  }
}

// New function: Capture full page using debugger API
async function captureFullPageWithDebugger(tabId) {
  try {
    // Attach debugger
    await chrome.debugger.attach({ tabId }, '1.3');

    // Enable Page domain
    await chrome.debugger.sendCommand({ tabId }, 'Page.enable');

    // Set transparent background (optional, helps with white canvas)
    await chrome.debugger.sendCommand({ tabId }, 'Emulation.setDefaultBackgroundColorOverride', {
      color: { r: 0, g: 0, b: 0, a: 0 }
    });

    // Get full content height
    const { contentSize } = await chrome.debugger.sendCommand({ tabId }, 'Page.getLayoutMetrics');
    const { height, width } = contentSize;

    // Override viewport to full size
    await chrome.debugger.sendCommand({ tabId }, 'Emulation.setDeviceMetricsOverride', {
      height: height,
      width: width,
      deviceScaleFactor: 1,
      mobile: false
    });

    // Capture full screenshot
    const { data } = await chrome.debugger.sendCommand({ tabId }, 'Page.captureScreenshot', {
      format: 'png',
      fromSurface: true
    });

    const dataUrl = `data:image/png;base64,${data}`;

    // Clean up
    await chrome.debugger.sendCommand({ tabId }, 'Emulation.clearDeviceMetricsOverride');
    await chrome.debugger.detach({ tabId });

    return dataUrl;
  } catch (error) {
    console.error('[SnapToAI] Debugger capture failed:', error);
    try {
      await chrome.debugger.detach({ tabId });
    } catch {}
    return null;
  }
}
