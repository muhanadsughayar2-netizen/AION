// 1. THE API CALL (Hybrid Request)
async function askGemini(prompt, base64Image) {
  try {
    const response = await model.generateContent({
      contents: [{
        role: 'user',
        parts: [
          { text: prompt },
          // Only add image if it exists
          ...(base64Image ? [{ inlineData: { mimeType: "image/jpeg", data: base64Image } }] : []) 
        ]
      }],
      generationConfig: {
        // We ASK for audio, but the code handles it if we don't get it
        responseModalities: ["AUDIO"], 
        speechConfig: {
          voiceConfig: {
            prebuiltVoiceConfig: { voiceName: "Kore" }
          }
        }
      }
    });

    const candidate = response.response.candidates[0].content.parts[0];

    // 2. THE DECISION ENGINE (Audio vs Text)
    if (candidate.inlineData && candidate.inlineData.mimeType.startsWith("audio")) {
      // CASE A: We got Premium Native Audio
      console.log("playing native Gemini audio...");
      playNativeAudio(candidate.inlineData.data);
    } else {
      // CASE B: We only got Text (Fallback)
      console.log("Native audio unavailable, using fallback...");
      const text = candidate.text || "I analyzed the data but could not generate audio.";
      speakTextFallback(text);
    }
    
    return candidate.text; // Return text to display on screen

  } catch (error) {
    console.error("Gemini Error:", error);
    speakTextFallback("I am having trouble connecting to the financial network.");
  }
}

// 3. HELPER: Play Gemini's Premium Audio
function playNativeAudio(base64Audio) {
  const audioSrc = `data:audio/wav;base64,${base64Audio}`;
  const audio = new Audio(audioSrc);
  audio.play();
}

// 4. HELPER: The Backup (Browser Voice)
function speakTextFallback(text) {
  window.speechSynthesis.cancel(); // Stop any previous speech
  const utterance = new SpeechSynthesisUtterance(text);
  
  // Try to find a "Google" or "Natural" voice on the computer
  const voices = window.speechSynthesis.getVoices();
  const bestVoice = voices.find(v => v.name.includes("Google") || v.name.includes("Natural"));
  if (bestVoice) utterance.voice = bestVoice;
  
  window.speechSynthesis.speak(utterance);
}
