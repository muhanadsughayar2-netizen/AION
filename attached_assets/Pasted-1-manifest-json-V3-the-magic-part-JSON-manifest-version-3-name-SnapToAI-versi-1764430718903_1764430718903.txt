1. manifest.json (V3) – the magic part
JSON{
  "manifest_version": 3,
  "name": "SnapToAI",
  "version": "1.0",
  ...

  "host_permissions": [
    "https://grok.x.ai/*",
    "https://chatgpt.com/*",
    "https://claude.ai/*",
    "https://*.openai.com/*",
    "<all_urls>"
  ],

  "permissions": [
    "activeTab",
    "scripting",
    "storage",
    "desktopCapture"
  ],

  "web_accessible_resources": [
    {
      "resources": ["overlay.html", "overlay.js", "overlay.css"],
      "matches": ["<all_urls>"]
    }
  ],

  "background": {
    "service_worker": "background.js"
  },

  "content_scripts": [
    {
      "matches": ["<all_urls>"],
      "js": ["content.js"],
      "run_at": "document_start",
      "all_frames": true
    }
  ]
}
Key lines you probably missing:
"all_frames": true → survives Grok’s 17-layer iframe hell
"<all_urls>" in host_permissions + web_accessible_resources → stops CSP from killing your overlay on the second snap
2. background.js – re-inject on every snap request
JavaScriptchrome.runtime.onMessage.addListener((msg, sender, sendResponse) => {
  if (msg.action === "snap") {
    chrome.tabs.query({ active: true, currentWindow: true }, (tabs) => {
      const tabId = tabs[0].id;

      // Force re-inject the overlay every single time (this is the real fix)
      chrome.scripting.executeScript({
        target: { tabId: tabId, allFrames: true },
        files: ["content.js"]
      }, () => {
        setTimeout(() => {
          chrome.tabs.sendMessage(tabId, { action: "startSnap" });
        }, 150); // tiny delay so DOM is ready
      });
    });
  }
});
3. content.js – tiny but bulletproof overlay injector
JavaScriptfunction injectOverlay() {
  if (document.getElementById('snaptoai-overlay')) return;

  const div = document.createElement('div');
  div.id = 'snaptoai-overlay';
  div.innerHTML = `
    <div style="position:fixed;top:0;left:0;width:100vw;height:100vh;z-index:999999;background:rgba(0,0,0,0.3);cursor:crosshair;" id="snaptoai-selector">
      <div style="position:absolute;top:10px;left:10px;color:white;background:#0008;padding:8px 12px;border-radius:8px;font-family:sans-serif;">
        Drag to select • ESC to cancel
      </div>
    </div>`;
  document.body.appendChild(div);

  // your existing drag-to-select logic here
  // when user finishes → send canvas to background → stitch → send to Grok/ChatGPT
}

chrome.runtime.onMessage.addListener((msg) => {
  if (msg.action === "startSnap") {
    injectOverlay();
  }
});
