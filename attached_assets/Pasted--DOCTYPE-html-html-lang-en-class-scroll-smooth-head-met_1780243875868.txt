  <script>
    // Initialize Lucide icons
    lucide.createIcons();

    // Reveal Animation Observer
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
        }
      });
    }, { threshold: 0.1 });
    document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

    // Tab switching logic
    function showTab(id, btn) {
      document.querySelectorAll('.tab-panel').forEach(p => p.classList.remove('active'));
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
      const target = document.getElementById(id);
      if (target) target.classList.add('active');
      if (btn) btn.classList.add('active');
    }

    // Bio Score simulation logic
    let currentScore = 40;
    function toggleBioModule(btn, val) {
      btn.classList.toggle('active');
      if (btn.classList.contains('active')) {
        currentScore += val;
      } else {
        currentScore -= val;
      }
      updateScore();
    }

    function updateScore() {
      const text = document.getElementById('scoreText');
      const ring = document.getElementById('scoreRing');
      if (text && ring) {
        text.innerText = currentScore;
        const offset = 283 - (283 * currentScore / 100);
        ring.style.strokeDashoffset = offset;
      }
    }

    // Modal control logic
    function openLegal(title, body) {
      const modal = document.getElementById('legalModal');
      const mTitle = document.getElementById('modalTitle');
      const mBody = document.getElementById('modalBody');
      if (modal && mTitle && mBody) {
        mTitle.innerText = title;
        mBody.innerHTML = body;
        modal.style.display = 'flex';
        document.body.style.overflow = 'hidden';
      }
    }

    function closeLegal() {
      const modal = document.getElementById('legalModal');
      if (modal) {
        modal.style.display = 'none';
        document.body.style.overflow = 'auto';
      }
    }

    // Video Upload Logic
    function loadUserVideo() {
      const input = document.getElementById('videoUrlInput');
      const player = document.getElementById('mainVideoPlayer');
      let url = input.value.trim();
      
      if (!url) return;

      let videoId = '';
      if (url.includes('v=')) {
        videoId = url.split('v=')[1].split('&')[0];
      } else if (url.includes('youtu.be/')) {
        videoId = url.split('youtu.be/')[1].split('?')[0];
      } else if (url.includes('embed/')) {
        videoId = url.split('embed/')[1].split('?')[0];
      }

      if (videoId) {
        player.src = `https://www.youtube.com/embed/${videoId}?autoplay=1`;
        lucide.createIcons();
      } else {
        alert('Please enter a valid YouTube URL to analyze.');
      }
    }

    // Initialize display values
    updateScore();
  </script>
