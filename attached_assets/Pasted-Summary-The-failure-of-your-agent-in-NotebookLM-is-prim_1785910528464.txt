async function smartClick(targetText) {
  const findInShadows = (root, text) => {
    const elements = root.querySelectorAll('button, [role="button"], span, div, a');
    for (const el of elements) {
      const content = (el.innerText || el.getAttribute('aria-label') || '').toLowerCase();
      if (content.includes(text.toLowerCase())) return el;
      if (el.shadowRoot) {
        const found = findInShadows(el.shadowRoot, text);
        if (found) return found;
      }
    }
    return null;
  };

  const btn = findInShadows(document, targetText);
  if (btn) {
    btn.scrollIntoView({ behavior: 'smooth', block: 'center' });
    ['mousedown', 'mouseup', 'click'].forEach(type => {
      btn.dispatchEvent(new MouseEvent(type, { bubbles: true, view: window }));
    });
    return true;
  }
  return false;
}

async function fastInject(value) {

Summary
The critical failure in the NotebookLM workflow was a State Synchronization Gap. While the agent was successfully inserting text into the input fields, the underlying frameworks (Lit, Angular, and React) used by Google were not "noticing" the change because Input.insertText bypasses the standard DOM event loop. This left the "Insert" or "Generate" buttons in a disabled (greyed-out) state.

The fix involves a Force-Sync Event Dispatch that targets both the active element and its Shadow Host immediately after text insertion, ensuring the UI state updates and buttons become clickable.

Key Findings: The Framework "Blind Spot"
Shadow Host Isolation: In NotebookLM, the actual <input> or [contenteditable] is often deep inside a Shadow Root. Even if text is placed there, the "Host" element (the one the framework listens to) doesn't receive the bubble-up event.
Synthetic Input Events: Modern frameworks rely on onInput and onChange listeners. CDP's insertText is too "low-level" for these.
The Solution: After insertion, the agent must manually trigger a sequence of events: input $\rightarrow$ change $\rightarrow$ blur.

async function fastInject(value) {
  const input = document.querySelector('input[type="url"]') || 
                document.querySelector('textarea') || 
                document.activeElement;
                
  if (input) {
    input.focus();
    input.value = value;

    // Trigger the framework state update
    const eventConfig = { bubbles: true, composed: true, cancelable: true };
    input.dispatchEvent(new InputEvent('input', eventConfig));
    input.dispatchEvent(new Event('change', eventConfig));
    
    // Pierce back up to the Shadow Host if applicable
    const host = input.getRootNode().host;
    if (host) {
      host.dispatchEvent(new InputEvent('input', eventConfig));
      host.dispatchEvent(new Event('change', eventConfig));
    }
    
    console.log("State sync forced for NotebookLM.");
    return true;
  }
  return false;
}

let agentGoal = { total: 0, current: 0, active: false };
let stepRetryLimit = 2; // Reduced from 60 to 2
let currentStepRetries = 0;

async function agentSpeak(text) {
    window.speechSynthesis.cancel();
    const msg = new SpeechSynthesisUtterance(text);
    window.speechSynthesis.speak(msg);

async function processAgentStep(action, params) {
    // Check for Emergency Stop
    const { agentStopRequested } = await chrome.storage.session.get('agentStopRequested');
    if (agentStopRequested) {
        agentSpeak("Stopping immediately as requested.");
        return { aborted: true };
    }

    try {
        const result = await executeAction(action, params);
        
        if (result.success) {
            currentStepRetries = 0; // Reset on success
            // Check if this was the final goal
            if (agentGoal.active && action === 'insert_source') {
                agentGoal.current++;
                if (agentGoal.current >= agentGoal.total) {
                    agentSpeak(`I've finished! I added all ${agentGoal.total} videos for you. What's next?`);
                    agentGoal.active = false;
                }
            }
            return result;
        } else {
            currentStepRetries++;
            if (currentStepRetries >= stepRetryLimit) {
                // Strike Two: Ask for help
                agentSpeak("I'm stuck. I can't find the button to continue. Could you please click it for me? I'll watch and resume once you do.");
                return await requestUserIntervention(action);
            }
            return { retry: true };
        }
    } catch (e) {
        return { success: false, error: e.message };
    }
}

function findButtonAggressive(text) {
    const isMatch = (el) => (el.innerText || el.getAttribute('aria-label') || '').toLowerCase().includes(text.toLowerCase());
    
    // Recursive Piercer
    const pierce = (root) => {
        const els = root.querySelectorAll('button, [role="button"], a, span');
        for (const el of els) {
            if (isMatch(el)) return el;
            if (el.shadowRoot) {
                const found = pierce(el.shadowRoot);
                if (found) return found;
            }
        }
        return null;
    };
    return pierce(document);
}
