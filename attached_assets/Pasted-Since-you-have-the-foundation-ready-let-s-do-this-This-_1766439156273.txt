🎙️ Step 3: Add the "Mouth" (Text-to-Speech)
To make it "Live," add this simple function. Call it right after Gemini returns the text response.

JavaScript

function speak(text) {
  const synth = window.speechSynthesis;
  const utterance = new SpeechSynthesisUtterance(text);
  
  // High-quality voice selection
  const voices = synth.getVoices();
  utterance.voice = voices.find(v => v.name.includes('Google') && v.lang === 'en-US') || voices[0];
  
  // Visual feedback on your Star
  const star = document.getElementById('star-mic');
  utterance.onstart = () => star.classList.add('glowing-gold');
  utterance.onend = () => star.classList.remove('glowing-gold');

  synth.speak(utterance);
}
